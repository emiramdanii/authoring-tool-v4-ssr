    13→    13→const LONG_TEXT = 'X'.repeat(600);

    15→    15→
    16→    16→/**
    17→    17→ * Safely call renderQuizBlock and return { result, error }.
    18→    18→ * Does NOT use expect().not.toThrow so we can capture both outcomes.
    19→    19→ */
    20→    20→function safeRender(type: string, block: Record<string, unknown>) {
    21→    21→  try {
    22→    22→    const result = renderQuizBlock(type, block, noopRender);
    23→    23→    return { result, error: null as Error | null };
    24→    24→  } catch (e) {
    25→    25→    return { result: null as string | null, error: e as Error };
    26→    26→  }
    27→    27→}
