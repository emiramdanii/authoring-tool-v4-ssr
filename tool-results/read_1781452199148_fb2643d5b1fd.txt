     2→// EXPORT SERIALIZATION BOUNDARY — Sprint 6.4-E1-QA2
     3→// ═══════════════════════════════════════════════════════════════════════
     4→// Tests the JSON → HTML <script> injection boundary for ALL export routes.
     5→//
     6→// The production export path injects project data as:
     7→//   <script>window.__EXPORT_DATA__=${dataJson};</script>
     8→//
     9→// This boundary is the most critical security checkpoint:
    10→// if JSON.stringify + escaping is insufficient, a payload can
    11→// terminate the </script> tag and inject arbitrary HTML/JS.
    12→//
    13→// OWASP Reference: XSS Prevention Cheat Sheet — Rule #3
    14→// "Escape the following characters with HTML encoding"
    15→// in the context of HTML script content:
    16→//   < → \u003c    > → \u003e    / → \u002f
    17→//   & → \u0026    U+2028 → \u2028    U+2029 → \u2029
    18→// ═══════════════════════════════════════════════════════════════════════
    19→
    20→import { describe, it, expect } from 'vitest';

    25→// ═══════════════════════════════════════════════════════════════════════
    26→// SECTION A: SERIALIZATION FUNCTION — Test the canonical serializer
    27→// ═══════════════════════════════════════════════════════════════════════
    28→
    29→/**
    30→ * Legacy serialization (what was used before QA2).
    31→ * Kept for comparison — should behave identically to the canonical
    32→ * serializer for <, >, / escaping, but is MISSING & and U+2028/U+2029.
    33→ */
    34→function serializeForScriptLegacy(data: unknown): string {
    35→  return JSON.stringify(data)
    36→    .replace(/</g, '\\u003c')
    37→    .replace(/>/g, '\\u003e')
    38→    .replace(/\//g, '\\u002f');
    39→}
