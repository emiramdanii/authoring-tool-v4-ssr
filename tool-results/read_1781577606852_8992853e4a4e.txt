    91→function getBlockIcon(blockType: string, blockIcon: string): React.ReactNode {
    92→  // Use capability registry as single source of truth for interactive blocks
    93→  if (BlockCapabilityRegistry.get(blockType).derived.interactive) {
    94→    return <span className="material-symbols-outlined text-emerald-400" style={ { fontSize: '16px' } }>sports_esports</span>;

    99→  return <span className="text-sm leading-none">{blockIcon}</span>;

   102→// ═══════════════════════════════════════════════════════════════════
   103→// FUZZY SEARCH
   104→// ═══════════════════════════════════════════════════════════════════
   105→
   106→function fuzzyScore(query: string, text: string): number {
   107→  const q = query.toLowerCase();
   108→  const t = text.toLowerCase();
   109→
   110→  // Exact match gets highest score
   111→  if (t === q) return 1000;
   112→
   113→  // Starts with query gets high score
   114→  if (t.startsWith(q)) return 500;
   115→
   116→  // Contains query as substring
   117→  const idx = t.indexOf(q);
   118→  if (idx !== -1) return 300 - idx;
   119→
   120→  // Fuzzy match — count how many chars matched in order
   121→  let qi = 0;
   122→  let lastMatchIdx = -1;
   123→  for (let ti = 0; ti < t.length && qi < q.length; ti++) {
   124→    if (t[ti] === q[qi]) {
   125→      qi++;
   126→      lastMatchIdx = ti;
   127→    }
   128→  }
   129→  if (qi === q.length) return 100 - lastMatchIdx;
   130→
   131→  return 0;
   132→}

   149→
   150→const RECENT_KEY = 'mpi-command-palette-recent';
   151→const MAX_RECENT = 8;
   152→
   153→function getRecentItems(): string[] {
   154→  try {
   155→    const data = localStorage.getItem(RECENT_KEY);
   156→    return data ? JSON.parse(data) : [];
   157→  } catch {
   158→    return [];
   159→  }
   160→}
   161→
   162→function addRecentItem(id: string) {
   163→  try {
   164→    const recent = getRecentItems().filter(r => r !== id);
   165→    recent.unshift(id);
   166→    localStorage.setItem(RECENT_KEY, JSON.stringify(recent.slice(0, MAX_RECENT)));
   167→  } catch {
   168→    // Ignore storage errors
   169→  }
   170→}

   533→      const page = pages[i];
   534→      commands.push({
   535→        id: `nav-page-${i}`,
   536→        label: `Ke ${page!.label || `Halaman ${i + 1}`}`,
   537→        description: `Pergi ke halaman ${i + 1}`,
   538→        icon: <span className="material-symbols-outlined text-purple-400" style={ { fontSize: '16px' } }>tag</span>,
   539→        category: 'navigation',
   540→        action: () => useCanvaStore.getState().goPage(i),
   541→      });
   542→    }
   543→
   544→    return commands;
   545→  }, [open]); // Rebuild when palette opens to get fresh state
   546→
   547→  // ── Filter and sort commands ───────────────────────────────
   548→  const filteredCommands = useMemo(() => {
   549→    if (!query.trim()) return allCommands;
   550→
   551→    const results = allCommands
   552→      .map(cmd => ({
   553→        cmd,
   554→        score: Math.max(
   555→          fuzzyScore(query, cmd.label),
   556→          cmd.description ? fuzzyScore(query, cmd.description) * 0.5 : 0,
   557→        ),
   558→      }))
   559→      .filter(r => r.score > 0)
   560→      .sort((a, b) => b.score - a.score)
   561→      .map(r => r.cmd);
   562→
   563→    return results;
   564→  }, [query, allCommands]);
   565→
   566→  // ── Group commands by category ─────────────────────────────
   567→  const groupedCommands = useMemo(() => {
   568→    // Show recently used at top when no query
   569→    const recentIds = query.trim() ? [] : getRecentItems();
   570→    const recentItems = recentIds
   571→      .map(id => allCommands.find(c => c.id === id))
   572→      .filter(Boolean) as CommandItem[];
   573→
   574→    // Group by category in display order
   575→    const categoryOrder: CommandCategory[] = ['block', 'game', 'action', 'navigation'];
   576→    const groups: { category: CommandCategory | 'recent'; items: CommandItem[] }[] = [];
   577→
   578→    // Add recent group if any
   579→    if (recentItems.length > 0) {
   580→      groups.push({ category: 'recent', items: recentItems });
   581→    }
   582→
   583→    for (const cat of categoryOrder) {
   584→      const items = filteredCommands.filter(c => c.category === cat);
   585→      if (items.length > 0) {
   586→        groups.push({ category: cat, items });
   587→      }
   588→    }
   589→
   590→    return groups;
   591→  }, [filteredCommands, allCommands, query]);
   592→
   593→  // ── Flat list for keyboard navigation ──────────────────────
   594→  const flatItems = useMemo(() => {
   595→    return groupedCommands.flatMap(g => g.items);
   596→  }, [groupedCommands]);
   597→
   598→  // ── Reset on open ──────────────────────────────────────────
   599→  // When the palette opens, reset search and selection state.
   600→  // Using requestAnimationFrame to avoid the set-state-in-effect lint rule
   601→  // while still ensuring the reset happens before the user interacts.
   602→  useEffect(() => {
   603→    if (open) {
   604→      const raf = requestAnimationFrame(() => {
   605→        setQuery('');
   606→        setSelectedIndex(0);
   607→        inputRef.current?.focus();
   608→      });
   609→      return () => cancelAnimationFrame(raf);
   610→    }
   611→    return undefined;
   612→  }, [open]);
   613→
   614→  // ── Clamp selected index (derived, no effect) ───────────────
   615→  const clampedSelectedIndex = selectedIndex >= flatItems.length
   616→    ? Math.max(0, flatItems.length - 1)
   617→    : selectedIndex;
   618→
   619→  // ── Scroll selected item into view ─────────────────────────
   620→  useEffect(() => {
   621→    if (listRef.current) {
   622→      const selected = listRef.current.querySelector(`[data-command-index="${clampedSelectedIndex}"]`);
   623→      if (selected) {
   624→        selected.scrollIntoView({ block: 'nearest' });
   625→      }
   626→    }
   627→  }, [clampedSelectedIndex]);
   628→
   629→  // ── Execute command ────────────────────────────────────────
   630→  const executeCommand = useCallback((cmd: CommandItem) => {
   631→    addRecentItem(cmd.id);
   632→    cmd.action();
   633→    onClose();
   634→  }, [onClose]);
   635→
   636→  // ── Keyboard handler ───────────────────────────────────────
   637→  const handleKeyDown = useCallback((e: React.KeyboardEvent) => {
   638→    switch (e.key) {
   639→      case 'ArrowDown':
   640→        e.preventDefault();
   641→        setSelectedIndex(i => (i + 1) % Math.max(1, flatItems.length));
   642→        break;
   643→      case 'ArrowUp':
   644→        e.preventDefault();
   645→        setSelectedIndex(i => (i - 1 + flatItems.length) % Math.max(1, flatItems.length));
   646→        break;
   647→      case 'Enter':
   648→        e.preventDefault();
   649→        if (flatItems[clampedSelectedIndex]) {
   650→          executeCommand(flatItems[clampedSelectedIndex]);
   651→        }
   652→        break;
   653→      case 'Escape':
   654→        e.preventDefault();
   655→        onClose();
   656→        break;
   657→    }
   658→  }, [flatItems, clampedSelectedIndex, executeCommand, onClose]);
   659→
   660→  // ── Don't render if not open ───────────────────────────────
   661→  if (!open) return null;
   662→
   663→  return createPortal(
   664→    <div
   665→      className="fixed inset-0 z-[100] flex items-start justify-center pt-[15vh]"
   666→      onClick={(e) => { if (e.target === e.currentTarget) onClose(); }}
   667→      role="dialog"
   668→      aria-modal="true"
   669→      aria-label="Command Palette"
   670→    >
   671→      {/* Backdrop */}
   672→      <div className="absolute inset-0 bg-black/50 backdrop-blur-sm" />
   673→
   674→      {/* Palette */}
   675→      <div
   676→        className={cn(
   677→          'relative w-full max-w-[520px] rounded-xl border border-app-border shadow-2xl overflow-hidden',
   678→          'bg-app-surface/95 backdrop-blur-md',
   679→          'animate-in fade-in zoom-in-95 duration-150',
   680→        )}
   681→        onKeyDown={handleKeyDown}
   682→      >
   683→        {/* Search Input */}
   684→        <div className="flex items-center gap-2 px-4 py-3 border-b border-app-border">
   685→          <span className="material-symbols-outlined text-app-muted shrink-0" style={ { fontSize: '18px' } }>search</span>
   686→          <input
   687→            ref={inputRef}
   688→            type="text"
   689→            value={query}
   690→            onChange={(e) => { setQuery(e.target.value); setSelectedIndex(0); }}
   691→            placeholder="Cari block, aksi, atau navigasi..."
   692→            className="flex-1 bg-transparent text-sm text-app-primary placeholder:text-app-muted outline-none"
   693→            aria-label="Cari perintah"
   694→            autoComplete="off"
   695→          />
   696→          <kbd className="hidden sm:inline-flex items-center gap-0.5 px-1.5 py-0.5 text-[10px] font-mono text-app-muted bg-app-elevated rounded border border-app-border">
   697→            Esc
   698→          </kbd>
   699→        </div>
   700→
   701→        {/* Results */}
   702→        <div
   703→          ref={listRef}
   704→          className="max-h-[420px] overflow-y-auto overscroll-contain py-1"
   705→          style={{
   706→            scrollbarWidth: 'thin',
   707→            scrollbarColor: 'rgba(255,255,255,0.1) transparent',
   708→          }}
   709→        >
   710→          {flatItems.length === 0 ? (
   711→            <div className="px-4 py-8 text-center text-sm text-app-muted">
   712→              <span className="material-symbols-outlined mx-auto mb-2 opacity-30" style={ { fontSize: '24px' } }>search</span>
   713→              Tidak ditemukan untuk &ldquo;{query}&rdquo;
   714→            </div>
   715→          ) : (
   716→            groupedCommands.map((group, groupIdx) => {
   717→              // Find the starting index for this group in the flat list
   718→              let startIdx = 0;
   719→              for (let g = 0; g < groupIdx; g++) {
   720→                startIdx += groupedCommands[g]!.items.length;
   721→              }
   722→
   723→              return (
   724→                <div key={group.category} role="group" aria-label={group.category === 'recent' ? 'Terakhir Digunakan' : CATEGORY_CONFIG[group.category]?.label}>
   725→                  {/* Group heading */}
   726→                  <div className="flex items-center gap-2 px-3 py-1.5">
   727→                    <span className="text-[10px] font-bold uppercase tracking-wider text-app-muted">
   728→                      {group.category === 'recent' ? '⏱ Terakhir Digunakan' : CATEGORY_CONFIG[group.category]?.label}
   729→                    </span>
   730→                    {group.category !== 'recent' && (
   731→                      <span className={cn(
   732→                        'inline-flex items-center px-1.5 py-0 rounded text-[9px] font-semibold border',
   733→                        CATEGORY_CONFIG[group.category].badgeClass,
   734→                      )}>
   735→                        {CATEGORY_CONFIG[group.category].badge}
   736→                      </span>
   737→                    )}
   738→                    <div className="flex-1 h-px bg-app-border/50" />
   739→                  </div>
   740→
   741→                  {/* Group items */}
   742→                  {group.items.map((cmd, itemIdx) => {
   743→                    const flatIdx = startIdx + itemIdx;
   744→                    const isSelected = flatIdx === clampedSelectedIndex;
   745→
   746→                    return (
   747→                      <div
   748→                        key={cmd.id}
   749→                        data-command-index={flatIdx}
   750→                        role="option"
   751→                        aria-selected={isSelected}
   752→                        onClick={() => executeCommand(cmd)}
   753→                        onMouseEnter={() => setSelectedIndex(flatIdx)}
   754→                        className={cn(
   755→                          'flex items-center gap-2.5 px-3 py-2 mx-1 rounded-lg cursor-pointer transition-colors',
   756→                          isSelected
   757→                            ? 'bg-app-accent/15 text-app-accent'
   758→                            : 'text-app-primary hover:bg-app-elevated',
   759→                        )}
   760→                      >
   761→                        {/* Icon */}
   762→                        <span className={cn(
   763→                          'flex items-center justify-center w-7 h-7 rounded-md shrink-0',
   764→                          isSelected ? 'bg-app-accent/20' : 'bg-app-elevated',
   765→                        )}>
   766→                          {cmd.icon}
   767→                        </span>
   768→
   769→                        {/* Text */}
   770→                        <div className="flex-1 min-w-0">
   771→                          <div className="text-[12px] font-medium truncate">{cmd.label}</div>
   772→                          {cmd.description && (
   773→                            <div className={cn(
   774→                              'text-[10px] truncate',
   775→                              isSelected ? 'text-app-accent/60' : 'text-app-muted',
   776→                            )}>
   777→                              {cmd.description}
   778→                            </div>
   779→                          )}
   780→                        </div>
   781→
   782→                        {/* Shortcut */}
   783→                        {cmd.shortcut && (
   784→                          <kbd className="hidden sm:inline-flex items-center gap-0.5 px-1.5 py-0.5 text-[9px] font-mono text-app-muted bg-app-elevated rounded border border-app-border/50">
   785→                            {cmd.shortcut}
   786→                          </kbd>
   787→                        )}
   788→
   789→                        {/* Category badge (small, for visual distinction) */}
   790→                        {group.category !== 'recent' && (
   791→                          <span className={cn(
   792→                            'shrink-0 inline-flex items-center px-1 py-0 rounded text-[8px] font-semibold border',
   793→                            CATEGORY_CONFIG[cmd.category].badgeClass,
   794→                          )}>
   795→                            {CATEGORY_CONFIG[cmd.category].badge}
   796→                          </span>
   797→                        )}
   798→                      </div>
   799→                    );
   800→                  })}
   801→                </div>
   802→              );
   803→            })
   804→          )}
   805→        </div>
   806→
   807→        {/* Footer hint */}
   808→        <div className="flex items-center gap-3 px-4 py-2 border-t border-app-border/50 text-[9px] text-app-muted">
   809→          <span className="flex items-center gap-1">
   810→            <kbd className="px-1 py-0.5 bg-app-elevated rounded border border-app-border/50 font-mono">↑↓</kbd>
   811→            navigasi
   812→          </span>
   813→          <span className="flex items-center gap-1">
   814→            <kbd className="px-1 py-0.5 bg-app-elevated rounded border border-app-border/50 font-mono">↵</kbd>
   815→            pilih
   816→          </span>
   817→          <span className="flex items-center gap-1">
   818→            <kbd className="px-1 py-0.5 bg-app-elevated rounded border border-app-border/50 font-mono">Esc</kbd>
   819→            tutup
   820→          </span>
   821→          <span className="ml-auto opacity-60">
   822→            {flatItems.length} perintah
   823→          </span>
   824→        </div>
   825→      </div>
   826→    </div>,
   827→    document.body,
   828→  );
   829→}

   834→// Manages the open/close state and Cmd+K keyboard shortcut.
   835→
   836→export function useCommandPalette() {
   837→  const [open, setOpen] = useState(false);
   838→
   839→  const toggle = useCallback(() => setOpen(v => !v), []);
   840→  const openPalette = useCallback(() => setOpen(true), []);
   841→  const closePalette = useCallback(() => setOpen(false), []);
   842→
   843→  // Register Cmd+K / Ctrl+K globally
   844→  useEffect(() => {
   845→    const handleKeyDown = (e: KeyboardEvent) => {
   846→      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
   847→        e.preventDefault();
   848→        e.stopPropagation();
   849→        toggle();
   850→      }
   851→    };
   852→    window.addEventListener('keydown', handleKeyDown, true); // Use capture phase to intercept before CanvaBuilder
   853→    return () => window.removeEventListener('keydown', handleKeyDown, true);
   854→  }, [toggle]);
   855→
   856→  // Listen for custom open event (from Toolbar or other components)
   857→  useEffect(() => {
   858→    const handleOpen = () => setOpen(true);
   859→    window.addEventListener(COMMAND_PALETTE_OPEN_EVENT, handleOpen);
   860→    return () => window.removeEventListener(COMMAND_PALETTE_OPEN_EVENT, handleOpen);
   861→  }, []);
   862→
   863→  return { open, setOpen, toggle, openPalette, closePalette };
   864→}
