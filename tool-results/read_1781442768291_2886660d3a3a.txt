   393→          setTimeout(function() {

   448→    function checkFillBlank(input) {
   449→      var userAns = input.value.trim().toLowerCase();
   450→      var correctAns = input.dataset.answer.toLowerCase();
   451→      // Support multiple accepted answers separated by /
   452→      var accepted = correctAns.split('/');
   453→      if (accepted.some(function(a) { return a.trim() === userAns; })) {
   454→        input.classList.add('correct');
   455→        input.classList.remove('wrong');
   456→      } else {
   457→        input.classList.add('wrong');
   458→        input.classList.remove('correct');
   459→      }
   460→    }
