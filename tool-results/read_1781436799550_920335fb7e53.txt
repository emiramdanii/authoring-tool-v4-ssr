    16→// ── Background-image-aware styling helpers ────────────────────────
    17→// When a background image is active, quiz options/feedback need more
    18→// opaque backgrounds and visible borders so they remain readable
    19→// on top of the image. When no bg image, original styles preserved.
    20→
    21→function kuisOptionBg(tokens: TokenResolver, isAnswered: boolean, isCorrect: boolean, isPicked: boolean): string {
    22→  if (!tokens.hasBackgroundImage()) {
    23→    if (!isAnswered) return tokens.subtleBg(0.04);
    24→    if (isCorrect) return tokens.accentBg('g', 0.08);
    25→    if (isPicked) return tokens.accentBg('r', 0.08);
    26→    return tokens.subtleBg(0.04);
    27→  }
    28→  const isDark = tokens.isDark();
    29→  if (!isAnswered) return isDark ? 'rgba(15,23,42,0.78)' : 'rgba(255,255,255,0.86)';
    30→  if (isCorrect) {
    31→    const tint = tokens.accentBg('g', 0.16);
    32→    return `linear-gradient(${tint}, ${tint}), ${isDark ? 'rgba(15,23,42,0.78)' : 'rgba(255,255,255,0.86)'}`;
    33→  }
    34→  if (isPicked) {
    35→    const tint = tokens.accentBg('r', 0.16);
    36→    return `linear-gradient(${tint}, ${tint}), ${isDark ? 'rgba(15,23,42,0.78)' : 'rgba(255,255,255,0.86)'}`;
    37→  }
    38→  return isDark ? 'rgba(15,23,42,0.68)' : 'rgba(255,255,255,0.76)';
    39→}
    40→
    41→function kuisOptionBorder(tokens: TokenResolver, isAnswered: boolean, isCorrect: boolean, isPicked: boolean): string {
    42→  if (!tokens.hasBackgroundImage()) {
    43→    if (!isAnswered) return tokens.subtleBorder(0.1);
    44→    if (isCorrect) return tokens.color('g');
    45→    if (isPicked) return tokens.color('r');
    46→    return tokens.subtleBorder(0.08);
    47→  }
    48→  const isDark = tokens.isDark();
    49→  if (!isAnswered) return isDark ? 'rgba(255,255,255,0.22)' : 'rgba(0,0,0,0.14)';
    50→  if (isCorrect) return tokens.color('g');
    51→  if (isPicked) return tokens.color('r');
    52→  return isDark ? 'rgba(255,255,255,0.12)' : 'rgba(0,0,0,0.08)';
    53→}
    54→
    55→function kuisFeedbackBg(tokens: TokenResolver, isCorrect: boolean): string {
    56→  if (!tokens.hasBackgroundImage()) return tokens.accentBg(isCorrect ? 'g' : 'r', 0.06);
    57→  const tint = tokens.accentBg(isCorrect ? 'g' : 'r', 0.16);
    58→  const isDark = tokens.isDark();
    59→  return `linear-gradient(${tint}, ${tint}), ${isDark ? 'rgba(15,23,42,0.82)' : 'rgba(255,255,255,0.88)'}`;
    60→}

   353→export const KuisRenderer = React.memo(function KuisRenderer({ block, tokens, interactive, isCompact, isEditing, pageIndex, compression }: {
   354→  block: KuisBlock; tokens: TokenResolver; interactive: boolean; isCompact: boolean; isEditing?: boolean; pageIndex?: number; compression?: import('../../layout/CompressionEngine').CompressionDecision;
   355→}) {
   356→  const [current, setCurrent] = React.useState(0);
   357→  const [answers, setAnswers] = React.useState<Record<number, number>>({});
   358→  const [showExplanation, setShowExplanation] = React.useState(false);
   359→
   360→  // ── Educational design tokens ──────────────────────────────
   361→  const edu = tokens.edu('kuis', isCompact);
   362→
   363→  // ── Variant state ───────────────────────────────────────────
   364→  // Sprint 2K.4: Persist variant changes to schema (same pattern as other renderers)
   365→  const updateSchemaBlock = useCanvaStore((s) => s.updateSchemaBlock);
   366→  const [currentVariant, setCurrentVariant] = useState<'A' | 'B' | 'C'>(
   367→    (block.variant as 'A' | 'B' | 'C') || 'A'
   368→  );
   369→  const variant = currentVariant;
   370→  const handleVariantChange = useCallback((v: 'A' | 'B' | 'C') => {
   371→    setCurrentVariant(v);
   372→    if (block.id) updateSchemaBlock(block.id, { variant: v });
   373→  }, [block.id, updateSchemaBlock]);
   374→
   375→  // ── Compression-aware rendering (step-reveal strategy) ──────
   376→  // Kuis naturally shows one question at a time (step-reveal).
   377→  // When compressed, we switch to the most compact variant (C "Ringkas")
   378→  // and hide decorative elements (streak badge, progress aurora).
   379→  const questions = block.questions || [];
   380→  const { isCompressed } = useBlockCompression({
   381→    compression,
   382→    totalItems: questions.length,
   383→  });
   384→  const effectiveVariant = isCompressed ? 'C' as const : variant;
   385→
   386→  // ── Replay watcher: reset all state when replayGeneration bumps ──
   387→  const replayGeneration = useInteractiveStore(s => s.replayGeneration);
   388→  React.useEffect(() => {
   389→    setCurrent(0);
   390→    setAnswers({});
   391→    setShowExplanation(false);
   392→  }, [replayGeneration]);
   393→
   394→  const q = questions[current];
   395→  const totalAnswered = React.useMemo(
   396→    () => Object.keys(answers).length,
   397→    [answers],
   398→  );
   399→  const totalCorrect = React.useMemo(
   400→    () => Object.entries(answers).filter(([idx, ans]) => questions[Number(idx)]?.ans === ans).length,
   401→    [answers, questions],
   402→  );
   403→  const isCompleted = totalAnswered >= questions.length && questions.length > 0;
   404→
   405→  // ── Streak calculation ──────────────────────────────────────
   406→  const currentStreak = React.useMemo(() => {
   407→    let s = 0;
   408→    for (let i = current - 1; i >= 0; i--) {
   409→      if (answers[i] !== undefined && questions[i]?.ans === answers[i]) s++;
   410→      else break;
   411→    }
   412→    return s;
   413→  }, [answers, current, questions]);
   414→
   415→  // ── Interactive store: score reporting ──────────────────────
   416→  const reportScore = useInteractiveStore(s => s.reportScore);
   417→
   418→  // Report score on completion (guard: only fire once per completion cycle)
   419→  const hasReportedRef = React.useRef(false);
   420→  React.useEffect(() => {
   421→    if (isCompleted && interactive && block.id && !hasReportedRef.current) {
   422→      hasReportedRef.current = true;
   423→      reportScore({
   424→        elementId: block.id,
   425→        pageIndex: pageIndex ?? 0,
   426→        score: totalCorrect * 20,
   427→        maxScore: questions.length * 20,
   428→        completed: true,
   429→      });
   430→      // Play tier-appropriate sound & confetti
   431→      const pct = Math.round((totalCorrect / questions.length) * 100);
   432→      if (pct >= 50) { playSound('complete'); }
   433→      else playSound('ding');
   434→    }
   435→    if (!isCompleted) hasReportedRef.current = false;
   436→  }, [isCompleted, interactive, block.id, totalCorrect, questions.length, reportScore, pageIndex]);
   437→
   438→  // ── Inline editing hooks ─────────────────────────────────────
   439→  const titleEditor = useInlineEditor({
   440→    blockId: block.id,
   441→    fieldKey: 'title',
   442→    value: block.title ?? '',
   443→    tag: 'span',
   444→  });
   445→  const questionEditor = useInlineEditor({
   446→    blockId: block.id,
   447→    fieldKey: `questions.${current}.q`,
   448→    value: q?.q ?? '',
   449→    tag: 'div',
   450→    multiline: true,
   451→  });
   452→  const explanationEditor = useInlineEditor({
   453→    blockId: block.id,
   454→    fieldKey: `questions.${current}.ex`,
   455→    value: q?.ex ?? '',
   456→    tag: 'span',
   457→  });
   458→
   459→  // ── Progress calculation ──────────────────────────────────────
   460→  const progress = questions.length > 0 ? totalAnswered / questions.length : 0;
   461→
   462→  // ── Answer handler (shared across variants) ───────────────────
   463→  const handleAnswer = React.useCallback(
   464→    (qCurrent: number, optIndex: number) => {
   465→      if (interactive && answers[qCurrent] === undefined) {
   466→        setAnswers(prev => ({ ...prev, [qCurrent]: optIndex }));
   467→        if (optIndex === questions[qCurrent]?.ans) { playSound('correct'); }
   468→        else playSound('incorrect');
   469→      }
   470→    },
   471→    [interactive, answers, questions],
   472→  );
   473→
   474→  // ══ COMPLETION SCREEN — Clean & Calm ═══════════════════════
   475→  if (isCompleted) {
   476→    const pct = questions.length > 0 ? Math.round((totalCorrect / questions.length) * 100) : 0;
   477→    const tierColor = pct >= 80 ? 'y' : pct >= 50 ? 'g' : 'o';
   478→
   479→    return (
   480→      <PremiumBlockWrapper tokens={tokens} accent={tierColor} staggerIndex={0}>
   481→        <ReadingProgressIndicator progress={1} tokens={tokens} accent={tierColor} height={2} position="top" />
   482→        <div className="text-center p-5" style={{ maxWidth: tokens.narrowWidth(), margin: '0 auto', ...(edu.hasBackgroundImage() ? edu.cardStyle() : {}) }}>
   483→
   484→          {/* Score pill */}
   485→          <div className="inline-flex items-center gap-2 px-4 py-2 rounded-full mb-4"
   486→            style={{
   487→              background: tokens.accentBg(tierColor, 0.08),
   488→              border: `1px solid ${tokens.colorAlpha(tierColor, 0.2)}`,
   489→            }}>
   490→            <span style={{ ...edu.title(), color: tokens.color(tierColor) }}>{pct}%</span>
   491→            <span style={{ ...edu.caption(), color: edu.mutedText(0.65) }}>{totalCorrect}/{questions.length}</span>
   492→          </div>
