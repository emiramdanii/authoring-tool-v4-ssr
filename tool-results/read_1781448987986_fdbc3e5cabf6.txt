   467→    function checkFillBlank(input) {

   470→      // Support multiple accepted answers separated by /
   471→      var accepted = correctAns.split('/');
   472→      if (accepted.some(function(a) { return a.trim() === userAns; })) {
   473→        input.classList.add('correct');
   474→        input.classList.remove('wrong');
   475→      } else {
   476→        input.classList.add('wrong');
   477→        input.classList.remove('correct');
   478→      }
   479→    }
