   135→// BOTTOM NAVIGATION — 56px fixed height, contract-aware
   136→// (Same as LearningMediaShell but no "Back to Edit")

   138→
   139→function ExportBottomNav({
   140→  currentScreen,
   141→  totalScreens,
   142→  pageStatuses,
   143→  templateType,
   144→  onNext,
   145→  onPrev,
   146→  onGoToScreen,
   147→  isNextLocked,
   148→  lockReason,
   149→  isDark,
   150→}: {
   151→  currentScreen: number;
   152→  totalScreens: number;
   153→  pageStatuses: PageCompletionStatus[];
   154→  templateType: string;
   155→  onNext: () => void;
   156→  onPrev: () => void;
   157→  onGoToScreen: (index: number) => void;
   158→  isNextLocked: boolean;
   159→  lockReason: string;
   160→  isDark: boolean;
   161→}) {
   162→  // Cover page gets a prominent "Mulai" button
   163→  const isCoverPage = currentScreen === 0 && templateType === 'cover';
   164→
   165→  // Completion indicator for each dot
   166→  const getDotClass = (index: number) => {
   167→    const isCurrent = index === currentScreen;
   168→    const status = pageStatuses[index];
   169→
   170→    if (isCurrent) {
   171→      return 'w-6 h-2.5 bg-blue-500';
   172→    }
   173→
   174→    switch (status) {
   175→      case 'completed':
   176→        return 'w-2.5 h-2.5 bg-emerald-400 hover:bg-emerald-500';
   177→      case 'locked':
   178→        return 'w-2.5 h-2.5 bg-amber-400 hover:bg-amber-500 ring-1 ring-amber-300';
   179→      case 'incomplete':
   180→      default:
   181→        return 'w-2.5 h-2.5 bg-slate-300 hover:bg-slate-400';
   182→    }
   183→  };
   184→
   185→  return (
   186→    <div className={`h-14 flex items-center justify-between px-4 shrink-0 z-10 ${
   187→      isDark
   188→        ? 'bg-[#0e1c2f]/95 backdrop-blur-md border-t border-white/10'
   189→        : 'bg-white border-t border-slate-200'
   190→    }`}>
   191→      {/* Prev — hidden on cover page */}
   192→      {!isCoverPage ? (
   193→        <Button
   194→          variant="ghost"
   195→          size="sm"
   196→          onClick={onPrev}
   197→          disabled={currentScreen <= 0}
   198→          className="flex items-center gap-1 px-3 py-2 rounded-lg text-sm font-medium disabled:opacity-30 active:scale-95 transition-transform"
   199→        >
   200→          <ChevronLeft size={18} />
   201→          <span className="hidden sm:inline">Sebelumnya</span>
   202→        </Button>
   203→      ) : (
   204→        <div className="w-24" />
   205→      )}
   206→
   207→      {/* Screen dots — with completion indicators */}
   208→      <div className="flex items-center gap-1.5 overflow-x-auto max-w-[60%] px-2">
   209→        {Array.from({ length: totalScreens }, (_, i) => {
   210→          const status = pageStatuses[i];
   211→          const statusIcon = status === 'completed' ? ' ✓' : status === 'locked' ? ' 🔒' : '';
   212→          return (
   213→            <button
   214→              key={i}
   215→              onClick={() => onGoToScreen(i)}
   216→              className={`
   217→                shrink-0 rounded-full transition-all duration-300
   218→                ${getDotClass(i)}
   219→              `}
   220→              aria-label={`Halaman ${i + 1}${statusIcon}`}
   221→              title={`Halaman ${i + 1}${statusIcon}`}
   222→            />
   223→          );
   224→        })}
   225→      </div>
   226→
   227→      {/* Mulai / Next / Selesai / Terkunci — context-aware */}
   228→      {isCoverPage ? (
   229→        /* ── MULAI button — prominent CTA on cover page ── */
   230→        <Button
   231→          size="sm"
   232→          onClick={onNext}
   233→          className="flex items-center gap-2 px-5 py-2.5 rounded-xl text-sm font-bold bg-emerald-500 hover:bg-emerald-600 text-white shadow-md shadow-emerald-500/25 active:scale-95 transition-all"
   234→        >
   235→          <Play size={16} />
   236→          <span>Mulai</span>
   237→        </Button>
   238→      ) : currentScreen >= totalScreens - 1 ? (
   239→        /* ── SELESAI button — on last page ── */
   240→        <Button
   241→          size="sm"
   242→          onClick={onNext}
   243→          className="flex items-center gap-1 px-3 py-2 rounded-lg text-sm font-medium bg-emerald-500 hover:bg-emerald-600 text-white"
   244→        >
   245→          <CheckCircle2 size={16} />
   246→          <span>Selesai</span>
   247→        </Button>
   248→      ) : isNextLocked ? (
   249→        /* ── TERKUNCI button — navigation locked by contract ── */
   250→        <Button
   251→          variant="ghost"
   252→          size="sm"
   253→          disabled
   254→          className="flex items-center gap-1 px-3 py-2 rounded-lg text-sm font-medium opacity-50 cursor-not-allowed"
   255→          title={lockReason}
   256→        >
   257→          <Lock size={14} className="text-amber-500" />
   258→          <span className="hidden sm:inline text-amber-600">Terkunci</span>
   259→        </Button>
   260→      ) : (
   261→        /* ── SELANJUTNYA button — normal next navigation ── */
   262→        <Button
   263→          variant="ghost"
   264→          size="sm"
   265→          onClick={onNext}
   266→          className="flex items-center gap-1 px-3 py-2 rounded-lg text-sm font-medium active:scale-95 transition-transform"
   267→        >
   268→          <span className="hidden sm:inline">Selanjutnya</span>
   269→          <ChevronRight size={18} />
   270→        </Button>
   271→      )}
   272→    </div>
   273→  );
   274→}

   279→
   280→function LockToast({
   281→  reason,
   282→  onDismiss,
   283→}: {
   284→  reason: string;
   285→  onDismiss: () => void;
   286→}) {
   287→  // Auto-dismiss after 4 seconds

   294→    <div className="fixed bottom-20 left-1/2 -translate-x-1/2 z-50 animate-in slide-in-from-bottom-4 fade-in duration-300">

   297→        <p className="text-sm text-amber-800 flex-1">{reason}</p>
   298→        <button
   299→          onClick={onDismiss}
   300→          className="text-amber-400 hover:text-amber-600 transition-colors shrink-0"

   314→function CompletionModal({
   315→  totalScore,
   316→  maxScore,
   317→  progress,
   318→  onRestart,
   319→  onClose,
   320→}: {
   321→  totalScore: number;

   331→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 backdrop-blur-sm">
   332→      <div className="bg-white rounded-2xl shadow-xl max-w-sm w-full mx-4 p-6 text-center relative">

   335→          onClick={onClose}

   344→            <Star
   345→              key={n}
   346→              size={32}
   347→              className={n <= starCount ? 'text-amber-400 fill-amber-400' : 'text-slate-200'}

   356→          Kamu telah menyelesaikan semua halaman pembelajaran.
   357→        </p>
   358→
   359→        {/* Score display */}
   360→        {maxScore > 0 && (
   361→          <div className="bg-amber-50 rounded-xl p-4 mb-4">
   362→            <div className="flex items-center justify-center gap-2 mb-1">
   363→              <Trophy size={20} className="text-amber-500" />

   380→              style={{ width: `${progress}%` }}
   381→            />
   382→          </div>
   383→        </div>
   384→
   385→        {/* Actions */}
   386→        <div className="flex gap-3">
   387→          <Button
   388→            variant="outline"
   389→            onClick={onRestart}
   390→            className="flex-1 flex items-center justify-center gap-2"

   425→// EXPORT APP — Main component
   426→// ═══════════════════════════════════════════════════════════════

   428→export default function ExportApp() {
   429→  const pages = useCanvaStore((s) => s.pages);
   430→  const ratioId = useCanvaStore((s) => s.ratioId);
   431→  const ratio = useCanvaStore((s) => {
   432→    const r = RATIOS.find(r => r.id === s.ratioId);
   433→    return r || RATIOS[0];
   434→  });
   435→  const meta = useAuthoringStore((s) => s.meta);
   436→
   437→  // Expose interactive store for Live Preview postMessage bridge
   438→  useEffect(() => {
   439→    window.__INTERACTIVE_STORE__ = useInteractiveStore;
   440→    return () => {
   441→      delete window.__INTERACTIVE_STORE__;
   442→    };
   443→  }, []);
   444→
   445→  // ── Learning media store — drives navigation, locks, completion ──
   446→  const currentScreenIndex = useLearningMediaStore(s => s.currentScreenIndex);
   447→  const totalScreens = useLearningMediaStore(s => s.totalScreens);
   448→  const showCompletionModal = useLearningMediaStore(s => s.showCompletionModal);
   449→  const showLockToast = useLearningMediaStore(s => s.showLockToast);
   450→  const navigationLockReason = useLearningMediaStore(s => s.navigationLockReason);
   451→  const initSession = useLearningMediaStore(s => s.initSession);
   452→  const nextScreen = useLearningMediaStore(s => s.nextScreen);
   453→  const prevScreen = useLearningMediaStore(s => s.prevScreen);
   454→  const forceGoToScreen = useLearningMediaStore(s => s.forceGoToScreen);
   455→  const getTotalScore = useLearningMediaStore(s => s.getTotalScore);
   456→  const getProgress = useLearningMediaStore(s => s.getProgress);
   457→  const showCompletion = useLearningMediaStore(s => s.showCompletion);
   458→  const dismissCompletion = useLearningMediaStore(s => s.dismissCompletion);
   459→  const dismissLockToast = useLearningMediaStore(s => s.dismissLockToast);
   460→  const resetSession = useLearningMediaStore(s => s.resetSession);
   461→  const canGoNext = useLearningMediaStore(s => s.canGoNext);
   462→  const getAllPageStatuses = useLearningMediaStore(s => s.getAllPageStatuses);
   463→
   464→  // ── Initialize learning session on mount ──
   465→  // Reset interactive-store scores so each export session starts fresh,
   466→  // then initialize the learning session with template types for contracts.
   467→  useEffect(() => {
   468→    // Clear stale scores from previous sessions
   469→    useInteractiveStore.getState().replayAll();
   470→
   471→    const templateTypes = pages.map(p => p.templateType || 'custom');
   472→    initSession(pages.length, templateTypes);
   473→  }, [pages.length, initSession]);
   474→
   475→  // Re-sync contracts when template types change
   476→  useEffect(() => {
   477→    const templateTypes = pages.map(p => p.templateType || 'custom');
   478→    const currentTypes = useLearningMediaStore.getState().templateTypes;
   479→    if (templateTypes.length !== currentTypes.length ||
   480→        templateTypes.some((t, i) => t !== currentTypes[i])) {
   481→      initSession(pages.length, templateTypes);
   482→    }
   483→  }, [pages, initSession]);
   484→
   485→  // ═══ Score sync bridge: interactive store → learning media store ═══
   486→  // When interactive widgets report scores via the interactive store,
   487→  // we bridge them to the learning media store for navigation locks
   488→  // and completion tracking. Same pattern as LearningMediaShell.
   489→  const syncScores = useLearningMediaStore(s => s.syncScores);
   490→  const markPageAnswered = useLearningMediaStore(s => s.markPageAnswered);
   491→  const markPageGameCompleted = useLearningMediaStore(s => s.markPageGameCompleted);
   492→  const markPageReflected = useLearningMediaStore(s => s.markPageReflected);
   493→
   494→  useEffect(() => {
   495→    const unsubscribe = useInteractiveStore.subscribe((state) => {
   496→      const completedScores = state.scores.filter(s => s.completed);
   497→      const entries = completedScores.map((s: InteractiveScoreEntry) => ({
   498→        pageId: s.elementId,
   499→        screenIndex: s.pageIndex,
   500→        score: s.score,
   501→        maxScore: s.maxScore,
   502→        timestamp: Date.now(),
   503→      }));
   504→      if (entries.length > 0) {
   505→        syncScores(entries);
   506→
   507→        // Trigger completion actions based on page contract type
   508→        const learnStore = useLearningMediaStore.getState();
   509→        for (const s of completedScores) {
   510→          const screenIndex = s.pageIndex;
   511→          const contract = learnStore.getContract(screenIndex);
   512→          if (contract) {
   513→            if (contract.completionType === 'answer') {
   514→              markPageAnswered(screenIndex);
   515→            } else if (contract.completionType === 'game') {
   516→              markPageGameCompleted(screenIndex);
   517→            } else if (contract.completionType === 'reflection') {
   518→              markPageReflected(screenIndex);
   519→            }
   520→          }
   521→        }
   522→      }
   523→    });
   524→
   525→    // Initial sync on mount
   526→    const initialScores = useInteractiveStore.getState().scores.filter(s => s.completed);
   527→    const initialEntries = initialScores.map((s: InteractiveScoreEntry) => ({
   528→      pageId: s.elementId,
   529→      screenIndex: s.pageIndex,
   530→      score: s.score,
   531→      maxScore: s.maxScore,
   532→      timestamp: Date.now(),
   533→    }));
   534→    if (initialEntries.length > 0) {
   535→      syncScores(initialEntries);
   536→
   537→      const learnStore = useLearningMediaStore.getState();
   538→      for (const s of initialScores) {
   539→        const screenIndex = s.pageIndex;
   540→        const contract = learnStore.getContract(screenIndex);
   541→        if (contract) {
   542→          if (contract.completionType === 'answer') markPageAnswered(screenIndex);
   543→          else if (contract.completionType === 'game') markPageGameCompleted(screenIndex);
   544→          else if (contract.completionType === 'reflection') markPageReflected(screenIndex);
   545→        }
   546→      }
   547→    }
   548→
   549→    return () => unsubscribe();
   550→  }, [syncScores, markPageAnswered, markPageGameCompleted, markPageReflected]);
   551→
   552→  // ── ResizeObserver for responsive scaling ──
   553→  const canvasRef = useRef<HTMLDivElement>(null);
   554→  const [scale, setScale] = useState(0.5);
   555→  const [direction, setDirection] = useState<PageDirection>(0);
   556→  const prevIdxRef = useRef(currentScreenIndex);
   557→
   558→  useEffect(() => {
   559→    const el = canvasRef.current;
   560→    if (!el) return;
   561→    const observer = new ResizeObserver(() => {
   562→      setScale(computeSceneScale(
   563→        { w: ratio!.w, h: ratio!.h },
   564→        { w: el.clientWidth || 800, h: el.clientHeight || 500 },
   565→        20,
   566→      ));
   567→    });
   568→    observer.observe(el);
   569→    return () => observer.disconnect();
   570→  }, [ratio]);
   571→
   572→  // Track direction for page transition
   573→  useEffect(() => {
   574→    if (currentScreenIndex > prevIdxRef.current) setDirection(1);
   575→    else if (currentScreenIndex < prevIdxRef.current) setDirection(-1);
   576→    prevIdxRef.current = currentScreenIndex;
   577→  }, [currentScreenIndex]);
   578→
   579→  // ── Touch/swipe support ──
   580→  useEffect(() => {
   581→    let startX = 0;
   582→    let startedOnInteractive = false;
   583→    const onTouchStart = (e: TouchEvent) => {
   584→      const target = e.target as HTMLElement;
   585→      startedOnInteractive = !!target.closest('textarea, input, [data-game], [data-quiz], [data-interactive], [contenteditable="true"]');
   586→      startX = e.changedTouches[0]!.screenX;
   587→    };
   588→    const onTouchEnd = (e: TouchEvent) => {
   589→      if (startedOnInteractive) return;
   590→      const dx = e.changedTouches[0]!.screenX - startX;
   591→      if (Math.abs(dx) > 50) {
   592→        if (dx < 0) nextScreen();
   593→        else prevScreen();
   594→      }
   595→    };
   596→    window.addEventListener('touchstart', onTouchStart, { passive: true });
   597→    window.addEventListener('touchend', onTouchEnd, { passive: true });
   598→    return () => {
   599→      window.removeEventListener('touchstart', onTouchStart);
   600→      window.removeEventListener('touchend', onTouchEnd);
   601→    };
   602→  }, [nextScreen, prevScreen]);
   603→
   604→  // ── Keyboard navigation ──
   605→  useEffect(() => {
   606→    const handler = (e: KeyboardEvent) => {
   607→      const tag = (e.target as HTMLElement)?.tagName;
   608→      if (tag === 'INPUT' || tag === 'TEXTAREA') return;
   609→      if (e.key === 'ArrowRight') { e.preventDefault(); nextScreen(); }
   610→      if (e.key === 'ArrowLeft') { e.preventDefault(); prevScreen(); }
   611→      if (e.key === ' ' || e.key === 'Enter') {
   612→        e.preventDefault();
   613→        // On last page, space/enter triggers completion
   614→        if (currentScreenIndex >= totalScreens - 1) {
   615→          handleSelesai();
   616→        } else {
   617→          nextScreen();
   618→        }
   619→      }
   620→    };
   621→    window.addEventListener('keydown', handler);
   622→    return () => window.removeEventListener('keydown', handler);
   623→  }, [nextScreen, prevScreen, currentScreenIndex, totalScreens]);
   624→
   625→  // ── Computed state ──
   626→  const { earned: totalScore, possible: maxScore } = getTotalScore();
   627→  const progress = getProgress();
   628→  const { allowed: isNextAllowed, reason: nextLockReason } = canGoNext();
   629→  const pageStatuses = useMemo(() => getAllPageStatuses(), [currentScreenIndex, getAllPageStatuses]);
   630→
   631→  const page = pages[currentScreenIndex];
   632→  const templateType = page?.templateType || 'custom';
   633→  const phaseMeta = getPhaseMeta(templateType);
   634→  const pageTitle = page?.label || phaseMeta.label || `Halaman ${currentScreenIndex + 1}`;
   635→
   636→  // Detect dark content (same logic as LearningMediaShell)
   637→  const isDarkContent = (() => {
   638→    const schemaThemeId = page?.schema?.themeId || (page?.templateData as Record<string, unknown>)?.schemaThemeId as string | undefined;
   639→    const lightThemes = ['ios-light', 'ios-warm', 'minimal', 'ocean-light', 'warm-light'];
   640→    if (lightThemes.includes(schemaThemeId || '')) return false;
   641→    return true;
   642→  })();
   643→
   644→  // ── Handlers ──
   645→  const handleSelesai = useCallback(() => {
   646→    fireConfettiCelebration();
   647→    showCompletion();
   648→  }, [showCompletion]);
   649→
   650→  const handleRestart = useCallback(() => {
   651→    useInteractiveStore.getState().replayAll();
   652→    resetSession();
   653→    const templateTypes = pages.map(p => p.templateType || 'custom');
   654→    initSession(pages.length, templateTypes);
   655→  }, [resetSession, initSession, pages]);
   656→
   657→  // ── No pages state ──
   658→  if (pages.length === 0) {
   659→    return (
   660→      <div className="min-h-screen bg-slate-900 text-white flex items-center justify-center">
   661→        <div className="text-center">
   662→          <Inbox size={36} className="text-white/60" />
   663→          <p className="text-white/60">Belum ada halaman untuk ditampilkan.</p>
   664→        </div>
   665→      </div>
   666→    );
   667→  }
   668→
   669→  if (!page) return null;
   670→
   671→  return (
   672→    <div className={`flex-1 flex flex-col h-screen overflow-hidden ${
   673→      isDarkContent ? 'bg-[#080f1a]' : 'bg-slate-100'
   674→    }`}>
   675→      {/* Top Navbar — title, progress, score */}
   676→      <ExportTopNavbar
   677→        title={pageTitle}
   678→        progress={progress}
   679→        totalScore={totalScore}
   680→        maxScore={maxScore}
   681→        isDark={isDarkContent}
   682→      />
   683→
   684→      {/* Phase badge row */}
   685→      <div className={`flex items-center gap-2 px-4 py-1.5 shrink-0 ${
   686→        isDarkContent
   687→          ? 'bg-[#0e1c2f]/80 border-b border-white/5'
   688→          : 'bg-white/80 border-b border-slate-100'
   689→      }`}>
   690→        <PhaseBadge templateType={templateType} />
   691→        <span className="text-[11px] text-slate-400">
   692→          Halaman {currentScreenIndex + 1} dari {totalScreens}
   693→        </span>
   694→      </div>
   695→
   696→      {/* Main content area — scaled page preview */}
   697→      <div
   698→        ref={canvasRef}
   699→        className="flex-1 min-h-0 flex items-center justify-center overflow-hidden"
   700→      >
   701→        <PageTransition
   702→          pageKey={`export-page-${currentScreenIndex}`}
   703→          direction={direction}
   704→          duration={0.3}
   705→          className="relative overflow-hidden shadow-lg shadow-black/20 ring-1 ring-slate-200/50 rounded-sm"
   706→          style={{
   707→            width: ratio!.w,
   708→            height: ratio!.h,
   709→            transform: `scale(${scale})`,
   710→            transformOrigin: 'center center',
   711→          }}
   712→        >
   713→          <CanvasErrorBoundary name="Export">
   714→            <PageRenderer
   715→              mode="export"
   716→              page={page}
   717→              currentPageIndex={currentScreenIndex}
   718→              totalPages={totalScreens}
   719→            />
   720→          </CanvasErrorBoundary>
   721→        </PageTransition>
   722→      </div>
   723→
   724→      {/* Bottom Navigation — contract-aware */}
   725→      <ExportBottomNav
   726→        currentScreen={currentScreenIndex}
   727→        totalScreens={totalScreens}
   728→        pageStatuses={pageStatuses}
   729→        templateType={templateType}
   730→        onPrev={prevScreen}
   731→        onNext={currentScreenIndex >= totalScreens - 1 ? handleSelesai : nextScreen}
   732→        onGoToScreen={forceGoToScreen}
   733→        isNextLocked={!isNextAllowed && currentScreenIndex < totalScreens - 1}
   734→        lockReason={nextLockReason}
   735→        isDark={isDarkContent}
   736→      />
   737→
   738→      {/* Navigation Lock Toast */}
   739→      {showLockToast && navigationLockReason && (
   740→        <LockToast
   741→          reason={navigationLockReason}
   742→          onDismiss={dismissLockToast}
   743→        />
   744→      )}
   745→
   746→      {/* Completion Modal */}
   747→      {showCompletionModal && (
   748→        <CompletionModal
   749→          totalScore={totalScore}
   750→          maxScore={maxScore}
   751→          progress={progress}
   752→          onRestart={handleRestart}
   753→          onClose={dismissCompletion}
   754→        />
   755→      )}
   756→    </div>
   757→  );
   758→}
