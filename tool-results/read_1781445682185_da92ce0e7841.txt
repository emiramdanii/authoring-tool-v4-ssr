   466→    function checkFillBlank(input) {

   470→      var accepted = correctAns.split('/');

   472→        input.classList.add('correct');
   473→        input.classList.remove('wrong');
   474→      } else {
   475→        input.classList.add('wrong');
   476→        input.classList.remove('correct');
   477→      }
   478→    }
