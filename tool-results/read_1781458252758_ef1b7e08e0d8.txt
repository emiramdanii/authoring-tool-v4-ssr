   111→// ── Main Function ──────────────────────────────────────────────
   112→
   113→/**
   114→ * Apply a guided schema patch — the SINGLE write path to schema content.
   115→ *
   116→ * This function:
   117→ *   1. Finds the target page and block by ID
   118→ *   2. Deep-merges the patch into the block
   119→ *   3. Pushes undo history
   120→ *   4. Emits edit bus event
   121→ *   5. Validates document purity (dev mode)
   122→ *   6. Optionally handles overflow (auto-split)
   123→ *
   124→ * Usage from Konten tabs:
   125→ *   applyGuidedSchemaPatch({
   126→ *     pageId: currentPageId,
   127→ *     blockId: 'norma-golden-5',
   128→ *     patch: { questions: [newQuestion] },
   129→ *     source: 'konten-tab',
   130→ *   })
   131→ *
   132→ * Usage from Right Panel:
   133→ *   applyGuidedSchemaPatch({
   134→ *     pageId: currentPageId,
   135→ *     blockId: selectedBlockId,
   136→ *     patch: { title: 'New Title', subtitle: 'New Sub' },
   137→ *     source: 'guided-form',
   138→ *   })
   139→ */
   140→export function applyGuidedSchemaPatch(args: GuidedPatchArgs): GuidedPatchResult {
   141→  // Phase 4: Konten tab edits default to 'warn' overflow policy.
   142→  // This ensures teachers see overflow warnings when adding content
   143→  // that exceeds page capacity, without blocking the edit.
   144→  const resolvedOverflowPolicy = args.overflowPolicy
   145→    ?? (args.source === 'konten-tab' || args.source === 'dokumen-tab' ? 'warn' : 'none');
   146→  const { pageId, blockId, patch, overflowPolicy = resolvedOverflowPolicy, source = 'user' } = args;
   147→
   148→  const store = useCanvaStore.getState();
   149→  const pages = [...store.pages];
   150→  const pageIndex = pages.findIndex(p => p.id === pageId);
   151→
   152→  if (pageIndex < 0) {
   153→    return { success: false, error: `Page "${pageId}" not found`, pageId, blockId };
   154→  }
   155→
   156→  let page = pages[pageIndex]!;
   157→  let schema = ensurePageSchema(page);
   158→
   159→  if (!schema) {
   160→    return { success: false, error: `Page "${pageId}" has no schema`, pageId, blockId };
   161→  }
   162→
   163→  // Ensure page has schema (migrate if needed)
   164→  if (!page.schema && schema) {
   165→    page = { ...page, schema };
   166→    pages[pageIndex] = page;
   167→  }
   168→
   169→  const blocks = schema.blocks;
   170→  if (!Array.isArray(blocks)) {
   171→    return { success: false, error: `Page "${pageId}" schema has no blocks array`, pageId, blockId };
   172→  }
   173→
   174→  // Find the target block
   175→  const owner = findBlockOwner(blocks, blockId);
   176→  if (!owner) {
   177→    return { success: false, error: `Block "${blockId}" not found in page "${pageId}"`, pageId, blockId };
   178→  }
   179→
   180→  // Phase 4: Defer history push until after overflow check for 'reject' policy.
   181→  // For other policies, push before mutation as usual.
   182→  // If overflowPolicy === 'reject', we push history only after confirming the patch is accepted.
   183→  const shouldDeferHistory = overflowPolicy === 'reject';
   184→  if (!shouldDeferHistory) {
   185→    store._pushHistory();
   186→  }
   187→
   188→  // Apply the patch based on block ownership (top-level vs nested)
   189→  let newBlocks: SchemaBlock[];
   190→  let forwardPatches: import('immer').Patch[];
   191→  let inversePatches: import('immer').Patch[];
   192→  let blockType: string;
   193→
   194→  if (owner.kind === 'top-level') {
   195→    blockType = blocks[owner.index]!.type;
   196→
   197→    // Deep merge the patch into the target block using the existing merge utility
   198→    const mergeResult = mergeBlockInArray(blocks, owner.index, patch);
   199→    newBlocks = mergeResult.blocks;
   200→    forwardPatches = mergeResult.patches;
   201→    inversePatches = mergeResult.inversePatches;
   202→  } else {
   203→    // Nested block (materi-section content, ftab tab, children)
   204→    const result = produceWithPatches(blocks, draft => {
   205→      let target: SchemaBlock | undefined;
   206→      if (owner.kind === 'ftab-tab') {
   207→        const ft = draft[owner.blockIndex] as { tabs?: Array<{ content?: SchemaBlock[] }> };
   208→        target = ft.tabs?.[owner.tabIndex]?.content?.[owner.childIndex];
   209→      } else if (owner.kind === 'materi-section') {
   210→        const ms = draft[owner.blockIndex] as { content?: SchemaBlock[] };
   211→        target = ms.content?.[owner.childIndex];
   212→      } else if (owner.kind === 'children') {
   213→        target = draft[owner.blockIndex]!.children?.[owner.childIndex];
   214→      }
   215→      if (target) {
   216→        Object.assign(target, deepMergeBlock(target, patch));
   217→      }
   218→    });
   219→
   220→    newBlocks = result[0] as SchemaBlock[];
   221→    forwardPatches = result[1];
   222→    inversePatches = result[2];
   223→
   224→    // Determine block type for nested block
   225→    if (owner.kind === 'ftab-tab') {
   226→      const ft = blocks[owner.blockIndex] as { tabs?: Array<{ content?: SchemaBlock[] }> };
   227→      blockType = ft.tabs?.[owner.tabIndex]?.content?.[owner.childIndex]?.type || 'unknown';
   228→    } else if (owner.kind === 'materi-section') {
   229→      const ms = blocks[owner.blockIndex] as { content?: SchemaBlock[] };
   230→      blockType = ms.content?.[owner.childIndex]?.type || 'unknown';
   231→    } else {
   232→      blockType = blocks[owner.blockIndex]!.children?.[owner.childIndex]?.type || 'unknown';
   233→    }
   234→  }
   235→
   236→  // Commit the schema update
   237→  const newSchema = commitSchemaUpdate(schema, newBlocks);
   238→
   239→  // Dev-mode purity guard
   240→  assertDocumentPurity(newSchema, 'applyGuidedSchemaPatch');
   241→
   242→  // Write to store
   243→  pages[pageIndex] = { ...page, schema: newSchema };
   244→  useCanvaStore.setState({ pages });
   245→
   246→  // Phase 5: Mark dirty via useDirtyStore — any schema content edit means unsaved changes
   247→  try {
   248→    const { useDirtyStore } = require('@/store/dirty-store');
   249→    useDirtyStore.getState().markDirty();
   250→  } catch { /* SSR guard */ }
   251→
   252→  // Emit edit bus event for audit trail
   253→  editBus.emit({
   254→    type: 'patch',
   255→    patch: {
   256→      blockId,
   257→      blockType,
   258→      pageIndex,
   259→      patch,
   260→      timestamp: Date.now(),
   261→      source,
   262→      _immerPatches: { forward: forwardPatches, inverse: inversePatches, pageIndex },
   263→    },
   264→  });
   265→
   266→  // Handle overflow policy
   267→  let overflowDetected = false;
   268→  let autoSplitPerformed = false;
   269→  let newPageId: string | undefined;
   270→  let overflowDetails: OverflowCheckResult | undefined;
   271→
   272→  // Phase 4: For 'reject' policy, push history only after confirming the patch is accepted.
   273→  // This avoids ghost undo entries when a patch is rejected.
   274→  if (shouldDeferHistory) {
   275→    // Preview overflow before committing history
   276→    const previewCheck = checkOverflowRich(newSchema, page.templateType);
   277→    if (previewCheck.overflowDetected && overflowPolicy === 'reject') {
   278→      // Revert: restore original pages without pushing history
   279→      useCanvaStore.setState({ pages: store.pages });
   280→      return {
   281→        success: false,
   282→        error: 'Patch rejected: would cause content overflow',
   283→        overflowDetected: true,
   284→        pageId,
   285→        blockId,
   286→        overflowDetails: previewCheck,
   287→      };
   288→    }
   289→    // Patch accepted — now push history (snapshot = state BEFORE our mutation)
   290→    useCanvaStore.getState()._pushHistory();
   291→  }
   292→
   293→  if (overflowPolicy !== 'none') {
   294→    const check = checkOverflowRich(newSchema, page.templateType);
   295→    overflowDetected = check.overflowDetected;
   296→    overflowDetails = check;
   297→
   298→    if (overflowDetected) {
   299→      // Phase 4: Emit overflow warning to the UI store so Konten tabs
   300→      // and other components can show the OverflowWarningBanner.
   301→      useOverflowWarningStore.getState().setWarning({
   302→        pageId,
   303→        blockId,
   304→        source,
   305→        details: check,
   306→        timestamp: Date.now(),
   307→      });
   308→
   309→      if (overflowPolicy === 'warn') {
   310→        console.warn(
   311→          `[guided-patch] Overflow detected on page "${pageId}" after patching block "${blockId}". ` +
   312→          `${check.summary}`
   313→        );
   314→      } else if (overflowPolicy === 'reject') {
   315→        // Phase 4: This branch should no longer be reached — reject is handled
   316→        // above before history push. Keeping as safety net.
   317→        useCanvaStore.setState({ pages: store.pages });
   318→        return {
   319→          success: false,
   320→          error: 'Patch rejected: would cause content overflow',
   321→          overflowDetected: true,
   322→          pageId,
   323→          blockId,
   324→          overflowDetails: check,
   325→        };
   326→      } else if (overflowPolicy === 'auto-split') {
   327→        // Phase 4: Auto-split using promoteSceneSplitToPage() directly.
   328→        // This is ATOMIC — the split reads the current schema state (which already
   329→        // has our patched content) and creates a new page in a single transaction.
   330→        // No need to navigate first; promoteSceneSplitToPage() works by pageId.
   331→        // Phase 4.2: Recursive auto-split — keeps splitting while the original
   332→        // page still has overflow (up to 5 splits max to prevent infinite loops).
   333→        //
   334→        // BATCH UNDO: All splits in the loop are grouped under the single
   335→        // _pushHistory() call above. Each split internally calls _pushHistory(),
   336→        // so we set _skipHistory=true during the loop to prevent intermediate
   337→        // snapshots. The single undo of the original patch will restore the
   338→        // entire pre-split state.
   339→        if (check.canSplit && check.scenePlan && !check.scenePlan.isSingleScene) {
   340→          try {
   341→            const { promoteSceneSplitToPage } = require('./schema-apply');
   342→            const MAX_SPLITS = 5;
   343→            let currentCheck = check;
   344→            // Phase 4: Skip intermediate history pushes during auto-split loop
   345→            // so that a single undo reverts the entire patch + all splits
   346→            useCanvaStore.setState({ _skipHistory: true });
   347→            for (let i = 0; i < MAX_SPLITS; i++) {
   348→              const splitResult = promoteSceneSplitToPage(pageId, currentCheck.scenePlan!, 1);
   349→              if (splitResult.success && splitResult.pageUpdated) {
   350→                autoSplitPerformed = true;
   351→                if (!newPageId) newPageId = splitResult.newPageId; // Keep first new page ID
   352→              } else {
   353→                break; // Can't split further
   354→              }
   355→              // Check if the original page still overflows after this split
   356→              const { useCanvaStore } = require('@/store/canva/store');
   357→              const updatedPages = useCanvaStore.getState().pages;
   358→              const updatedPage = updatedPages.find((p: CanvaPage) => p.id === pageId);
   359→              if (!updatedPage?.schema) break;
   360→              const nextCheck = checkOverflowRich(updatedPage.schema, updatedPage.templateType);
   361→              if (!nextCheck.overflowDetected) break;
   362→              if (!nextCheck.canSplit || !nextCheck.scenePlan || nextCheck.scenePlan.isSingleScene) break;
   363→              currentCheck = nextCheck;
   364→            }
   365→            // Phase 4: Restore history tracking after auto-split loop
   366→            useCanvaStore.setState({ _skipHistory: false });
   367→          } catch (err) {
   368→            // Phase 4: Always restore history tracking, even on error
   369→            useCanvaStore.setState({ _skipHistory: false });
   370→            console.warn(
   371→              `[guided-patch] Auto-split failed for page "${pageId}": ${err}. ` +
   372→              `Content may overflow. Manual split recommended.`
   373→            );
   374→          }
   375→        } else if (!check.canSplit) {
   376→          console.warn(
   377→            `[guided-patch] Overflow on page "${pageId}" but page type "${page.templateType}" ` +
   378→            `does not allow splitting. Content may overflow. Consider shortening content.`
   379→          );
   380→        } else {
   381→          console.warn(
   382→            `[guided-patch] Overflow detected on page "${pageId}" but scene plan indicates ` +
   383→            `single scene. Content may overflow.`
   384→          );
   385→        }
   386→      }
   387→    } else {
   388→      // Phase 4: Auto-clear stale overflow status when content fits
   389→      // This ensures pageOverflowStatus is always accurate without
   390→      // requiring a manual scanAllPagesOverflow() call.
   391→      const overflowStore = useOverflowWarningStore.getState();
   392→      const currentStatus = overflowStore.pageOverflowStatus[pageId];
   393→      if (currentStatus?.hasOverflow) {
   394→        overflowStore.clearPageOverflowStatus(pageId);
   395→      }
   396→      // Also clear the banner if the warning was for this page
   397→      const currentWarning = overflowStore.lastWarning;
   398→      if (currentWarning?.pageId === pageId) {
   399→        overflowStore.clearWarning();
   400→      }
   401→    }
   402→  }
   403→
   404→  return {
   405→    success: true,
   406→    overflowDetected,
   407→    autoSplitPerformed,
   408→    newPageId,
   409→    pageId,
   410→    blockId,
   411→    overflowDetails,
   412→  };
   413→}

   415→// ── Batch Patch ────────────────────────────────────────────────
   416→
   417→/**
   418→ * Apply multiple patches atomically.
   419→ * All patches succeed or none are applied.
   420→ *
   421→ * Useful for: bulk edits from AI regenerate, multi-field updates from Konten tab.
   422→ */
   423→export function applyGuidedSchemaPatchBatch(
   424→  patches: GuidedPatchArgs[],
   425→): GuidedPatchResult[] {
   426→  const results: GuidedPatchResult[] = [];
   427→
   428→  // For now, apply sequentially. Transaction-based batch is a future enhancement.
   429→  for (const patchArgs of patches) {
   430→    const result = applyGuidedSchemaPatch(patchArgs);
   431→    results.push(result);
   432→
   433→    // If any patch fails, stop applying further patches
   434→    if (!result.success) {
   435→      break;
   436→    }
   437→  }
   438→
   439→  return results;
   440→}

   473→ * Get a specific block from a page's schema by block ID.

   476→export function getSchemaBlockById(pageId: string, blockId: string): SchemaBlock | null {
   477→  const store = useCanvaStore.getState();
   478→  const page = store.pages.find(p => p.id === pageId);

   485→/**
   486→ * Find a block in a page's blocks array (including nested containers).
   487→ */
   488→function findBlockByIdInPage(blocks: SchemaBlock[], blockId: string): SchemaBlock | null {
   489→  for (const block of blocks) {
   490→    if (block.id === blockId) return block;
   491→
   492→    // Check nested containers
   493→    const b = block as unknown as Record<string, unknown>;
   494→
   495→    // materi-section.content
   496→    if (b.content && Array.isArray(b.content)) {
   497→      const found = findBlockByIdInPage(b.content as SchemaBlock[], blockId);
   498→      if (found) return found;
   499→    }
   500→
   501→    // ftab.tabs[].content
   502→    if (b.tabs && Array.isArray(b.tabs)) {
   503→      for (const tab of b.tabs as Array<{ content?: SchemaBlock[] }>) {
   504→        if (tab.content) {
   505→          const found = findBlockByIdInPage(tab.content, blockId);
   506→          if (found) return found;
   507→        }
   508→      }
   509→    }
   510→
   511→    // children
   512→    if (block.children) {
   513→      const found = findBlockByIdInPage(block.children, blockId);
   514→      if (found) return found;
   515→    }
   516→  }
   517→
   518→  return null;
   519→}

   523→/**
   524→ * Field definition for the guided editor.
   525→ * Simpler than PropertyField — focused on CONTENT editing,
   526→ * not layout/style editing.
   527→ */
   528→export interface GuidedFieldDef {
   529→  /** Property key in the schema block */
   530→  key: string;
   531→  /** Display label (teacher-friendly) */
   532→  label: string;
   533→  /** Field type */
   534→  type: 'text' | 'textarea' | 'richtext' | 'color' | 'icon' | 'select' | 'number' | 'boolean' | 'array' | 'table-text';
   535→  /** Help text for teachers */
   536→  helpText?: string;
   537→  /** Placeholder */
   538→  placeholder?: string;
   539→  /** Is this field required? */
   540→  required?: boolean;
   541→  /** For array type: sub-field definitions */
   542→  fields?: GuidedFieldDef[];
   543→  /** For array type: max items */
   544→  maxItems?: number;
   545→  /** For select type: static options (use optionsFrom for dynamic options from block data) */
   546→  options?: Array<{ label: string; value: string }>;
   547→  /** For select type: dynamically populate options from another field in the block.
   548→   *  Reads block[optionsFrom.field] as an array, maps each item to
   549→   *  { label: item[optionsFrom.labelKey], value: item[optionsFrom.valueKey] }.
   550→   *  Falls back to `options` if the source array is empty or missing. */
   551→  optionsFrom?: { field: string; labelKey: string; valueKey: string };
   552→  /** For number type: min/max */
   553→  min?: number;
   554→  max?: number;
   555→  /** For array type: auto-generate an `id` field (nanoid) when adding new items.
   556→   *  Used when the schema type requires an `id` field on array items
   557→   *  (e.g. sortir-game kolom/pool) but the id should not be shown to teachers. */
   558→  autoId?: boolean;
   559→  /** For array type: how to display array items in the guided editor.
   560→   *  - 'expanded' (default): all items rendered vertically in a scrollable container
   561→   *  - 'tab': show a tab selector (1|2|3|...) and render only the active item.
   562→   *    Ideal for quiz questions where displaying all items would make the panel too long. */
   563→  displayMode?: 'expanded' | 'tab';
   564→  /** Conditional visibility: show this field only when another field's value is in `values`.
   565→   *  If undefined, the field is always shown (backward-compatible). */
   566→  showWhen?: { field: string; values: string[] };
   567→}
   568→
   569→/**
   570→ * Guided editor schema for a block type.
   571→ * Defines what CONTENT fields a teacher can edit.
   572→ * Does NOT include layout/style/position — those stay in the right panel.
   573→ */
   574→export interface GuidedEditorSchema {
   575→  /** Block type this schema belongs to */
   576→  blockType: string;
   577→  /** Display name */
   578→  displayName: string;
   579→  /** Description for teachers */
   580→  description: string;
   581→  /** Icon */
   582→  icon: string;
   583→  /** Ordered list of editable content fields */
   584→  fields: GuidedFieldDef[];
   585→  /** Sections for grouping fields into 3 zona: Isi Utama, Tampilan, Lanjutan */
   586→  sections?: Array<{
   587→    key: string;
   588→    label: string;
   589→    fieldKeys: string[];
   590→    /** Whether this section starts collapsed. Default: false (open).
   591→     *  Use true for "Tampilan" and "Lanjutan" zones. */
   592→    collapsed?: boolean;
   593→  }>;
   594→}
   595→
   596→/**
   597→ * Get the guided editor schema for a block type.
   598→ * Returns null if no guided editor is defined for this block type.
   599→ *
   600→ * Currently returns definitions for the most common block types.
   601→ * More block types will be added as Phase 2 progresses.
   602→ */
   603→export function getGuidedEditorSchema(blockType: string): GuidedEditorSchema | null {
   604→  return GUIDED_EDITOR_REGISTRY[blockType] ?? null;
   605→}
   606→
   607→/**
   608→ * Check if a block type has a guided editor defined.
   609→ */
   610→export function hasGuidedEditor(blockType: string): boolean {
   611→  return blockType in GUIDED_EDITOR_REGISTRY;
   612→}

  1434→// ── Overflow Detection ─────────────────────────────────────────
  1435→
  1436→/**
  1437→ * Rich overflow detection using SceneOverflowEngine.
  1438→ * Replaces the old word-count heuristic with real scene plan computation.
  1439→ *
  1440→ * Returns detailed overflow information including:
  1441→ *   - Whether overflow is detected
  1442→ *   - The computed scene plan
  1443→ *   - Whether the page type allows splitting
  1444→ *   - Whether compression is available
  1445→ *   - Total scenes needed
  1446→ *   - Human-readable summary
  1447→ */
  1448→export function checkOverflowRich(schema: ScreenSchema, templateType?: string): OverflowCheckResult {
  1449→  if (!schema.blocks || schema.blocks.length === 0) {
  1450→    return {
  1451→      overflowDetected: false,
  1452→      scenePlan: null,
  1453→      canSplit: false,
  1454→      canCompress: false,
  1455→      totalScenes: 1,
  1456→      summary: 'Tidak ada konten',
  1457→    };
  1458→  }
  1459→
  1460→  // ── Compute scene plan using real measurements ──
  1461→  // Read the project's actual ratioId instead of hardcoding '16:9'
  1462→  let ratioId = '16:9';
  1463→  try {
  1464→    const { useCanvaStore } = require('@/store/canva/store');
  1465→    ratioId = useCanvaStore.getState().ratioId || '16:9';
  1466→  } catch { /* fallback to 16:9 during SSR */ }
  1467→  const sceneRes = getSceneResolution(ratioId);
  1468→  const hasFullPageBlock = schema.blocks.length === 1 &&
  1469→    isFullPageBlockType(schema.blocks[0].type);
  1470→  const safeArea = hasFullPageBlock
  1471→    ? DEFAULT_SAFE_AREA
  1472→    : computeSafeArea({ showTopNav: false, showBottomNav: false, isCompact: true, pagePadding: 16 });
  1473→
  1474→  const scenePlan = computeScenePlan(schema, sceneRes, safeArea, { isCompact: true });
  1475→
  1476→  // ── Check contract canSplit ──
  1477→  const contract = getContractLazy()(undefined);
  1478→  const pageLayout = templateType
  1479→    ? contract.pageLayouts[templateType]
  1480→    : undefined;
  1481→  const canSplit = pageLayout?.canSplit ?? true; // Default: allow split
  1482→
  1483→  // ── Check if any overflowing blocks are compression-capable ──
  1484→  let canCompress = false;
  1485→  if (!scenePlan.isSingleScene) {
  1486→    for (const block of schema.blocks) {
  1487→      if (isBlockTypeCompressionCapable(block.type)) {
  1488→        canCompress = true;
  1489→        break;
  1490→      }
  1491→    }
  1492→  }
  1493→
  1494→  const overflowDetected = !scenePlan.isSingleScene;
  1495→
  1496→  // ── Human-readable summary ──
  1497→  let summary: string;
  1498→  if (!overflowDetected) {
  1499→    summary = 'Konten muat dalam satu halaman';
  1500→  } else {
  1501→    const scenesNeeded = scenePlan.totalScenes;
  1502→    const overflowBlocks = scenePlan.scenes
  1503→      .slice(1) // Blocks in scene 1+ overflow
  1504→      .reduce((acc, s) => acc + s.blockIds.length, 0);
  1505→    summary = `Konten melebihi kapasitas — butuh ${scenesNeeded} halaman (${overflowBlocks} blok overflow)`;
  1506→    if (!canSplit) {
  1507→      summary += '. Tipe halaman ini tidak bisa di-split — persingkat konten.';
  1508→    }
  1509→    if (canCompress) {
  1510→      summary += ' Kompresi tersedia.';
  1511→    }
  1512→  }
  1513→
  1514→  return {
  1515→    overflowDetected,
  1516→    scenePlan,
  1517→    canSplit,
  1518→    canCompress,
  1519→    totalScenes: scenePlan.totalScenes,
  1520→    summary,
  1521→  };
  1522→}

  1525→ * Simple boolean overflow check (backward-compatible wrapper).
  1526→ * Uses the rich check internally.
  1527→ */
  1528→export function checkOverflow(schema: ScreenSchema): boolean {
  1529→  return checkOverflowRich(schema).overflowDetected;

  1532→// ── Pre-Flight Overflow Preview ────────────────────────────────
  1533→
  1534→/**
  1535→ * Preview what would happen if a patch were applied — WITHOUT writing to the store.
  1536→ *
  1537→ * This is a "dry run" that:
  1538→ *   1. Clones the current page's schema
  1539→ *   2. Applies the patch to the clone
  1540→ *   3. Runs checkOverflowRich() on the cloned schema
  1541→ *   4. Returns the overflow result + the patched schema (for inspection)
  1542→ *   5. NEVER mutates the real store
  1543→ *
  1544→ * Use cases:
  1545→ *   - Konten tab: Show "This edit would cause overflow" BEFORE applying
  1546→ *   - Guided form: Pre-validate a field change
  1547→ *   - AI regenerate: Check if generated content fits before committing
  1548→ *
  1549→ * @param args - Same args as applyGuidedSchemaPatch
  1550→ * @returns Overflow check result + patched schema preview
  1551→ */
  1552→export function previewPatchOverflow(args: GuidedPatchArgs): {
  1553→  overflowCheck: OverflowCheckResult;
  1554→  /** The schema as it would look after the patch (read-only preview) */
  1555→  previewSchema: ScreenSchema | null;
  1556→} {
  1557→  const { pageId, blockId, patch } = args;
  1558→  const store = useCanvaStore.getState();
  1559→  const page = store.pages.find(p => p.id === pageId);
  1560→
  1561→  if (!page?.schema) {
  1562→    return {
  1563→      overflowCheck: {
  1564→        overflowDetected: false,
  1565→        scenePlan: null,
  1566→        canSplit: false,
  1567→        canCompress: false,
  1568→        totalScenes: 1,
  1569→        summary: 'Halaman tidak ditemukan atau tidak memiliki schema',
  1570→      },
  1571→      previewSchema: null,
  1572→    };
  1573→  }
  1574→
  1575→  // Clone the schema to avoid mutating the real store
  1576→  const schemaClone: ScreenSchema = JSON.parse(JSON.stringify(page.schema));
  1577→
  1578→  // Find the target block in the clone
  1579→  const owner = findBlockOwner(schemaClone.blocks, blockId);
  1580→  if (!owner) {
  1581→    return {
  1582→      overflowCheck: {
  1583→        overflowDetected: false,
  1584→        scenePlan: null,
  1585→        canSplit: false,
  1586→        canCompress: false,
  1587→        totalScenes: 1,
  1588→        summary: `Block "${blockId}" tidak ditemukan`,
  1589→      },
  1590→      previewSchema: null,
  1591→    };
  1592→  }
  1593→
  1594→  // Apply the patch to the clone (same logic as applyGuidedSchemaPatch)
  1595→  if (owner.kind === 'top-level') {
  1596→    const merged = deepMergeBlock(schemaClone.blocks[owner.index]!, patch);
  1597→    schemaClone.blocks[owner.index] = { ...schemaClone.blocks[owner.index], ...merged } as SchemaBlock;
  1598→  } else {
  1599→    // Nested block (materi-section content, ftab tab, children)
  1600→    let target: Record<string, unknown> | undefined;
  1601→    if (owner.kind === 'ftab-tab') {
  1602→      const ft = schemaClone.blocks[owner.blockIndex] as unknown as { tabs?: Array<{ content?: SchemaBlock[] }> };
  1603→      target = ft.tabs?.[owner.tabIndex]?.content?.[owner.childIndex] as Record<string, unknown> | undefined;
  1604→    } else if (owner.kind === 'materi-section') {
  1605→      const ms = schemaClone.blocks[owner.blockIndex] as unknown as { content?: SchemaBlock[] };
  1606→      target = ms.content?.[owner.childIndex] as Record<string, unknown> | undefined;
  1607→    } else if (owner.kind === 'children') {
  1608→      target = schemaClone.blocks[owner.blockIndex]!.children?.[owner.childIndex] as Record<string, unknown> | undefined;
  1609→    }
  1610→    if (target) {
  1611→      Object.assign(target, deepMergeBlock(target as SchemaBlock, patch));
  1612→    }
  1613→  }
  1614→
  1615→  // Run overflow check on the patched clone
  1616→  const overflowCheck = checkOverflowRich(schemaClone, page.templateType);
  1617→
  1618→  return {
  1619→    overflowCheck,
  1620→    previewSchema: schemaClone,
  1621→  };
  1622→}

  1624→// ── Convenience: Patch and get fresh projection ────────────────
  1625→
  1626→/**
  1627→ * Apply a guided schema patch and then derive a fresh projection
  1628→ * from the updated schema.
  1629→ *
  1630→ * This is the full cycle: write to schema → read projection back.
  1631→ * Used by Konten tabs that need to update the schema AND refresh
  1632→ * their local state from the new projection.
  1633→ *
  1634→ * @param args - Patch arguments
  1635→ * @param projectionDeriver - Function to derive projection from updated page
  1636→ * @returns Patch result + derived projection
  1637→ */
  1638→export function applyGuidedSchemaPatchWithProjection<T>(
  1639→  args: GuidedPatchArgs,
  1640→  projectionDeriver: (page: CanvaPage) => T,
  1641→): GuidedPatchResult & { projection: T | null } {
  1642→  const result = applyGuidedSchemaPatch(args);
  1643→
  1644→  if (!result.success) {
  1645→    return { ...result, projection: null };
  1646→  }
  1647→
  1648→  // Read the updated page from store
  1649→  const store = useCanvaStore.getState();
  1650→  const updatedPage = store.pages.find(p => p.id === result.pageId);
  1651→
  1652→  if (!updatedPage) {
  1653→    return { ...result, projection: null };
  1654→  }
  1655→
  1656→  const projection = projectionDeriver(updatedPage);
  1657→  return { ...result, projection };
  1658→}

  1660→// ── Batch Overflow Scan ─────────────────────────────────────────
  1661→
  1662→/**
  1663→ * Scan all pages for overflow and update the overflow warning store.
  1664→ *
  1665→ * This is the "post-generate overflow scan" — after auto-generate or
  1666→ * bulk content changes, run this to detect any pages that overflow.
  1667→ *
  1668→ * Results are written to useOverflowWarningStore.pageOverflowStatus
  1669→ * so the UI can show red dots on overflowing pages in the scene list,
  1670→ * export warnings, etc.
  1671→ *
  1672→ * @param options.autoSplit - If true, automatically split overflowing pages
  1673→ * @returns Summary of the scan
  1674→ */
  1675→export function scanAllPagesOverflow(options?: {
  1676→  autoSplit?: boolean;
  1677→}): {
  1678→  totalPages: number;
  1679→  overflowingPages: number;
  1680→  autoSplitResults: Array<{ pageId: string; success: boolean; newPageId?: string }>;
  1681→} {
  1682→  const store = useCanvaStore.getState();
  1683→  const pages = store.pages;
  1684→  const now = Date.now();
  1685→
  1686→  const statuses: Record<string, import('@/store/overflow-warning-store').PageOverflowStatus> = {};
  1687→  let overflowingPages = 0;
  1688→  const autoSplitResults: Array<{ pageId: string; success: boolean; newPageId?: string }> = [];
  1689→
  1690→  for (const page of pages) {
  1691→    if (!page.schema?.blocks || page.schema.blocks.length === 0) {
  1692→      statuses[page.id] = {
  1693→        hasOverflow: false,
  1694→        details: null,
  1695→        lastChecked: now,
  1696→      };
  1697→      continue;
  1698→    }
  1699→
  1700→    const check = checkOverflowRich(page.schema, page.templateType);
  1701→    statuses[page.id] = {
  1702→      hasOverflow: check.overflowDetected,
  1703→      details: check,
  1704→      lastChecked: now,
  1705→    };
  1706→
  1707→    if (check.overflowDetected) {
  1708→      overflowingPages++;
  1709→
  1710→      // Auto-split if requested and possible
  1711→      if (options?.autoSplit && check.canSplit && check.scenePlan && !check.scenePlan.isSingleScene) {
  1712→        try {
  1713→          const { promoteSceneSplitToPage } = require('./schema-apply');
  1714→          const splitResult = promoteSceneSplitToPage(page.id, check.scenePlan, 1);
  1715→          autoSplitResults.push({
  1716→            pageId: page.id,
  1717→            success: splitResult.success && splitResult.pageUpdated,
  1718→            newPageId: splitResult.newPageId,
  1719→          });
  1720→        } catch {
  1721→          autoSplitResults.push({ pageId: page.id, success: false });
  1722→        }
  1723→      }
  1724→    }
  1725→  }
  1726→
  1727→  // Write all statuses to the store in one batch
  1728→  useOverflowWarningStore.getState().batchSetPageOverflowStatus(statuses);
  1729→
  1730→  return {
  1731→    totalPages: pages.length,
  1732→    overflowingPages,
  1733→    autoSplitResults,
  1734→  };
  1735→}
