     9→// Noop render function matching the RenderBlockFn signature
    10→const noopRender = (() => '') as unknown as import('@/lib/export/utils').RenderBlockFn;
    11→
    12→// ── Helpers ──────────────────────────────────────────────────────────
    13→const LONG_TEXT = 'X'.repeat(600);

    15→
    16→/**
    17→ * Safely call renderQuizBlock and return { result, error }.
    18→ * Does NOT use expect().not.toThrow so we can capture both outcomes.
    19→ */
    20→function safeRender(type: string, block: Record<string, unknown>) {
    21→  try {
    22→    const result = renderQuizBlock(type, block, noopRender);
    23→    return { result, error: null as Error | null };
    24→  } catch (e) {
    25→    return { result: null as string | null, error: e as Error };
    26→  }
    27→}
