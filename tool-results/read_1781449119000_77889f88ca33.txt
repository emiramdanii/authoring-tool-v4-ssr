   241→   241→          size="sm"
   242→   242→          onClick={onNext}

   446→   446→  const currentScreenIndex = useLearningMediaStore(s => s.currentScreenIndex);
   447→   447→  const totalScreens = useLearningMediaStore(s => s.totalScreens);
   448→   448→  const showCompletionModal = useLearningMediaStore(s => s.showCompletionModal);
   449→   449→  const showLockToast = useLearningMediaStore(s => s.showLockToast);
   450→   450→  const navigationLockReason = useLearningMediaStore(s => s.navigationLockReason);
   451→   451→  const initSession = useLearningMediaStore(s => s.initSession);
   452→   452→  const nextScreen = useLearningMediaStore(s => s.nextScreen);
   453→   453→  const prevScreen = useLearningMediaStore(s => s.prevScreen);
   454→   454→  const forceGoToScreen = useLearningMediaStore(s => s.forceGoToScreen);
   455→   455→  const getTotalScore = useLearningMediaStore(s => s.getTotalScore);
   456→   456→  const getProgress = useLearningMediaStore(s => s.getProgress);
   457→   457→  const showCompletion = useLearningMediaStore(s => s.showCompletion);
   458→   458→  const dismissCompletion = useLearningMediaStore(s => s.dismissCompletion);
   459→   459→  const dismissLockToast = useLearningMediaStore(s => s.dismissLockToast);
   460→   460→  const resetSession = useLearningMediaStore(s => s.resetSession);
   461→   461→  const canGoNext = useLearningMediaStore(s => s.canGoNext);
   462→   462→  const getAllPageStatuses = useLearningMediaStore(s => s.getAllPageStatuses);
   463→   463→
   464→   464→  // ── Initialize learning session on mount ──
   465→   465→  // Reset interactive-store scores so each export session starts fresh,
   466→   466→  // then initialize the learning session with template types for contracts.
   467→   467→  useEffect(() => {
   468→   468→    // Clear stale scores from previous sessions
   469→   469→    useInteractiveStore.getState().replayAll();
   470→   470→
   471→   471→    const templateTypes = pages.map(p => p.templateType || 'custom');
   472→   472→    initSession(pages.length, templateTypes);
   473→   473→  }, [pages.length, initSession]);
   474→   474→
   475→   475→  // Re-sync contracts when template types change
   476→   476→  useEffect(() => {
   477→   477→    const templateTypes = pages.map(p => p.templateType || 'custom');
   478→   478→    const currentTypes = useLearningMediaStore.getState().templateTypes;
   479→   479→    if (templateTypes.length !== currentTypes.length ||
   480→   480→        templateTypes.some((t, i) => t !== currentTypes[i])) {
   481→   481→      initSession(pages.length, templateTypes);
   482→   482→    }
   483→   483→  }, [pages, initSession]);
   484→   484→
   485→   485→  // ═══ Score sync bridge: interactive store → learning media store ═══
   486→   486→  // When interactive widgets report scores via the interactive store,
   487→   487→  // we bridge them to the learning media store for navigation locks
   488→   488→  // and completion tracking. Same pattern as LearningMediaShell.
   489→   489→  const syncScores = useLearningMediaStore(s => s.syncScores);
   490→   490→  const markPageAnswered = useLearningMediaStore(s => s.markPageAnswered);
   491→   491→  const markPageGameCompleted = useLearningMediaStore(s => s.markPageGameCompleted);
   492→   492→  const markPageReflected = useLearningMediaStore(s => s.markPageReflected);
   493→   493→
   494→   494→  useEffect(() => {
   495→   495→    const unsubscribe = useInteractiveStore.subscribe((state) => {
   496→   496→      const completedScores = state.scores.filter(s => s.completed);
   497→   497→      const entries = completedScores.map((s: InteractiveScoreEntry) => ({
   498→   498→        pageId: s.elementId,
   499→   499→        screenIndex: s.pageIndex,
   500→   500→        score: s.score,
   501→   501→        maxScore: s.maxScore,
   502→   502→        timestamp: Date.now(),
   503→   503→      }));
   504→   504→      if (entries.length > 0) {
   505→   505→        syncScores(entries);
   506→   506→
   507→   507→        // Trigger completion actions based on page contract type
   508→   508→        const learnStore = useLearningMediaStore.getState();
   509→   509→        for (const s of completedScores) {
   510→   510→          const screenIndex = s.pageIndex;
   511→   511→          const contract = learnStore.getContract(screenIndex);
   512→   512→          if (contract) {
   513→   513→            if (contract.completionType === 'answer') {
   514→   514→              markPageAnswered(screenIndex);
   515→   515→            } else if (contract.completionType === 'game') {
   516→   516→              markPageGameCompleted(screenIndex);
   517→   517→            } else if (contract.completionType === 'reflection') {
   518→   518→              markPageReflected(screenIndex);
   519→   519→            }
   520→   520→          }
   521→   521→        }
   522→   522→      }
   523→   523→    });
   524→   524→
   525→   525→    // Initial sync on mount
   526→   526→    const initialScores = useInteractiveStore.getState().scores.filter(s => s.completed);
   527→   527→    const initialEntries = initialScores.map((s: InteractiveScoreEntry) => ({
   528→   528→      pageId: s.elementId,
   529→   529→      screenIndex: s.pageIndex,
   530→   530→      score: s.score,
   531→   531→      maxScore: s.maxScore,
   532→   532→      timestamp: Date.now(),
   533→   533→    }));
   534→   534→    if (initialEntries.length > 0) {
   535→   535→      syncScores(initialEntries);
   536→   536→
   537→   537→      const learnStore = useLearningMediaStore.getState();
   538→   538→      for (const s of initialScores) {
   539→   539→        const screenIndex = s.pageIndex;
   540→   540→        const contract = learnStore.getContract(screenIndex);
   541→   541→        if (contract) {
   542→   542→          if (contract.completionType === 'answer') markPageAnswered(screenIndex);
   543→   543→          else if (contract.completionType === 'game') markPageGameCompleted(screenIndex);
   544→   544→          else if (contract.completionType === 'reflection') markPageReflected(screenIndex);
   545→   545→        }
   546→   546→      }
   547→   547→    }
   548→   548→
   549→   549→    return () => unsubscribe();
   550→   550→  }, [syncScores, markPageAnswered, markPageGameCompleted, markPageReflected]);
   551→   551→
   552→   552→  // ── ResizeObserver for responsive scaling ──
   553→   553→  const canvasRef = useRef<HTMLDivElement>(null);
   554→   554→  const [scale, setScale] = useState(0.5);
   555→   555→  const [direction, setDirection] = useState<PageDirection>(0);
   556→   556→  const prevIdxRef = useRef(currentScreenIndex);
   557→   557→
   558→   558→  useEffect(() => {
   559→   559→    const el = canvasRef.current;
   560→   560→    if (!el) return;
   561→   561→    const observer = new ResizeObserver(() => {
   562→   562→      setScale(computeSceneScale(
   563→   563→        { w: ratio!.w, h: ratio!.h },
   564→   564→        { w: el.clientWidth || 800, h: el.clientHeight || 500 },
   565→   565→        20,
   566→   566→      ));
   567→   567→    });
   568→   568→    observer.observe(el);
   569→   569→    return () => observer.disconnect();
   570→   570→  }, [ratio]);
   571→   571→
   572→   572→  // Track direction for page transition
   573→   573→  useEffect(() => {
   574→   574→    if (currentScreenIndex > prevIdxRef.current) setDirection(1);
   575→   575→    else if (currentScreenIndex < prevIdxRef.current) setDirection(-1);
   576→   576→    prevIdxRef.current = currentScreenIndex;
   577→   577→  }, [currentScreenIndex]);
   578→   578→
   579→   579→  // ── Touch/swipe support ──
   580→   580→  useEffect(() => {
   581→   581→    let startX = 0;
   582→   582→    let startedOnInteractive = false;
   583→   583→    const onTouchStart = (e: TouchEvent) => {
   584→   584→      const target = e.target as HTMLElement;
   585→   585→      startedOnInteractive = !!target.closest('textarea, input, [data-game], [data-quiz], [data-interactive], [contenteditable="true"]');
   586→   586→      startX = e.changedTouches[0]!.screenX;
   587→   587→    };
   588→   588→    const onTouchEnd = (e: TouchEvent) => {
   589→   589→      if (startedOnInteractive) return;
   590→   590→      const dx = e.changedTouches[0]!.screenX - startX;
   591→   591→      if (Math.abs(dx) > 50) {
   592→   592→        if (dx < 0) nextScreen();
   593→   593→        else prevScreen();
   594→   594→      }
   595→   595→    };
   596→   596→    window.addEventListener('touchstart', onTouchStart, { passive: true });
   597→   597→    window.addEventListener('touchend', onTouchEnd, { passive: true });
   598→   598→    return () => {
   599→   599→      window.removeEventListener('touchstart', onTouchStart);
   600→   600→      window.removeEventListener('touchend', onTouchEnd);
   601→   601→    };
   602→   602→  }, [nextScreen, prevScreen]);
   603→   603→
   604→   604→  // ── Keyboard navigation ──
   605→   605→  useEffect(() => {
   606→   606→    const handler = (e: KeyboardEvent) => {
   607→   607→      const tag = (e.target as HTMLElement)?.tagName;
   608→   608→      if (tag === 'INPUT' || tag === 'TEXTAREA') return;
   609→   609→      if (e.key === 'ArrowRight') { e.preventDefault(); nextScreen(); }
   610→   610→      if (e.key === 'ArrowLeft') { e.preventDefault(); prevScreen(); }
   611→   611→      if (e.key === ' ' || e.key === 'Enter') {
   612→   612→        e.preventDefault();
   613→   613→        // On last page, space/enter triggers completion
   614→   614→        if (currentScreenIndex >= totalScreens - 1) {
   615→   615→          handleSelesai();
   616→   616→        } else {
   617→   617→          nextScreen();
   618→   618→        }
   619→   619→      }
   620→   620→    };
   621→   621→    window.addEventListener('keydown', handler);
   622→   622→    return () => window.removeEventListener('keydown', handler);
   623→   623→  }, [nextScreen, prevScreen, currentScreenIndex, totalScreens]);
   624→   624→
   625→   625→  // ── Computed state ──
   626→   626→  const { earned: totalScore, possible: maxScore } = getTotalScore();
   627→   627→  const progress = getProgress();
   628→   628→  const { allowed: isNextAllowed, reason: nextLockReason } = canGoNext();
   629→   629→  const pageStatuses = useMemo(() => getAllPageStatuses(), [currentScreenIndex, getAllPageStatuses]);
   630→   630→
   631→   631→  const page = pages[currentScreenIndex];
   632→   632→  const templateType = page?.templateType || 'custom';
   633→   633→  const phaseMeta = getPhaseMeta(templateType);
   634→   634→  const pageTitle = page?.label || phaseMeta.label || `Halaman ${currentScreenIndex + 1}`;
   635→   635→
   636→   636→  // Detect dark content (same logic as LearningMediaShell)
   637→   637→  const isDarkContent = (() => {
   638→   638→    const schemaThemeId = page?.schema?.themeId || (page?.templateData as Record<string, unknown>)?.schemaThemeId as string | undefined;
   639→   639→    const lightThemes = ['ios-light', 'ios-warm', 'minimal', 'ocean-light', 'warm-light'];
   640→   640→    if (lightThemes.includes(schemaThemeId || '')) return false;
   641→   641→    return true;
   642→   642→  })();
   643→   643→
   644→   644→  // ── Handlers ──
   645→   645→  const handleSelesai = useCallback(() => {
   646→   646→    fireConfettiCelebration();
   647→   647→    showCompletion();
   648→   648→  }, [showCompletion]);
   649→   649→
   650→   650→  const handleRestart = useCallback(() => {
   651→   651→    useInteractiveStore.getState().replayAll();
   652→   652→    resetSession();
   653→   653→    const templateTypes = pages.map(p => p.templateType || 'custom');
   654→   654→    initSession(pages.length, templateTypes);
   655→   655→  }, [resetSession, initSession, pages]);
   656→   656→
   657→   657→  // ── No pages state ──
   658→   658→  if (pages.length === 0) {
   659→   659→    return (
   660→   660→      <div className="min-h-screen bg-slate-900 text-white flex items-center justify-center">
   661→   661→        <div className="text-center">
   662→   662→          <Inbox size={36} className="text-white/60" />
   663→   663→          <p className="text-white/60">Belum ada halaman untuk ditampilkan.</p>
   664→   664→        </div>
   665→   665→      </div>
   666→   666→    );
   667→   667→  }
   668→   668→
   669→   669→  if (!page) return null;
   670→   670→
   671→   671→  return (
   672→   672→    <div className={`flex-1 flex flex-col h-screen overflow-hidden ${
   673→   673→      isDarkContent ? 'bg-[#080f1a]' : 'bg-slate-100'
   674→   674→    }`}>
   675→   675→      {/* Top Navbar — title, progress, score */}
   676→   676→      <ExportTopNavbar
   677→   677→        title={pageTitle}
   678→   678→        progress={progress}
   679→   679→        totalScore={totalScore}
   680→   680→        maxScore={maxScore}
   681→   681→        isDark={isDarkContent}
   682→   682→      />
   683→   683→
   684→   684→      {/* Phase badge row */}
   685→   685→      <div className={`flex items-center gap-2 px-4 py-1.5 shrink-0 ${
   686→   686→        isDarkContent
   687→   687→          ? 'bg-[#0e1c2f]/80 border-b border-white/5'
   688→   688→          : 'bg-white/80 border-b border-slate-100'
   689→   689→      }`}>
   690→   690→        <PhaseBadge templateType={templateType} />
   691→   691→        <span className="text-[11px] text-slate-400">
   692→   692→          Halaman {currentScreenIndex + 1} dari {totalScreens}
   693→   693→        </span>
   694→   694→      </div>
   695→   695→
   696→   696→      {/* Main content area — scaled page preview */}
   697→   697→      <div
   698→   698→        ref={canvasRef}
   699→   699→        className="flex-1 min-h-0 flex items-center justify-center overflow-hidden"
   700→   700→      >
   701→   701→        <PageTransition
   702→   702→          pageKey={`export-page-${currentScreenIndex}`}
   703→   703→          direction={direction}
   704→   704→          duration={0.3}
   705→   705→          className="relative overflow-hidden shadow-lg shadow-black/20 ring-1 ring-slate-200/50 rounded-sm"
   706→   706→          style={{
   707→   707→            width: ratio!.w,
   708→   708→            height: ratio!.h,
   709→   709→            transform: `scale(${scale})`,
   710→   710→            transformOrigin: 'center center',
   711→   711→          }}
   712→   712→        >
   713→   713→          <CanvasErrorBoundary name="Export">
   714→   714→            <PageRenderer
   715→   715→              mode="export"
   716→   716→              page={page}
   717→   717→              currentPageIndex={currentScreenIndex}
   718→   718→              totalPages={totalScreens}
   719→   719→            />
   720→   720→          </CanvasErrorBoundary>
   721→   721→        </PageTransition>
   722→   722→      </div>
   723→   723→
   724→   724→      {/* Bottom Navigation — contract-aware */}
   725→   725→      <ExportBottomNav
   726→   726→        currentScreen={currentScreenIndex}
   727→   727→        totalScreens={totalScreens}
   728→   728→        pageStatuses={pageStatuses}
   729→   729→        templateType={templateType}
   730→   730→        onPrev={prevScreen}
   731→   731→        onNext={currentScreenIndex >= totalScreens - 1 ? handleSelesai : nextScreen}
   732→   732→        onGoToScreen={forceGoToScreen}
   733→   733→        isNextLocked={!isNextAllowed && currentScreenIndex < totalScreens - 1}
   734→   734→        lockReason={nextLockReason}
   735→   735→        isDark={isDarkContent}
   736→   736→      />
   737→   737→
   738→   738→      {/* Navigation Lock Toast */}
   739→   739→      {showLockToast && navigationLockReason && (
   740→   740→        <LockToast
   741→   741→          reason={navigationLockReason}
   742→   742→          onDismiss={dismissLockToast}
   743→   743→        />
   744→   744→      )}
   745→   745→
   746→   746→      {/* Completion Modal */}
   747→   747→      {showCompletionModal && (
   748→   748→        <CompletionModal
   749→   749→          totalScore={totalScore}
   750→   750→          maxScore={maxScore}
   751→   751→          progress={progress}
   752→   752→          onRestart={handleRestart}
   753→   753→          onClose={dismissCompletion}
   754→   754→        />
   755→   755→      )}
   756→   756→    </div>
   757→   757→  );
   758→   758→}
