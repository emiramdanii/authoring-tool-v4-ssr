    30→// ── XSS Detection Helpers ───────────────────────────────────────────
    31→
    32→interface XssCheckResult {
    33→  safe: boolean;
    34→  violations: string[];
    35→}
    36→
    37→/**
    38→ * Check rendered HTML for XSS vulnerability indicators.
    39→ * Returns { safe, violations } — safe=true means no XSS patterns found.
    40→ */
    41→function checkXss(html: string, payload: string): XssCheckResult {
    42→  const violations: string[] = [];
    43→
    44→  // 1. Raw <script> tag in output (not escaped)
    45→  if (/<script[\s>]/i.test(html)) {
    46→    violations.push(`<script> tag found in output (unescaped)`);
    47→  }
    48→
    49→  // 2. Event handler attributes
    50→  const eventHandlers = ['onerror=', 'onclick=', 'onload=', 'ontoggle=', 'onmouseover='];
    51→  for (const handler of eventHandlers) {
    52→    if (html.toLowerCase().includes(handler)) {
    53→      violations.push(`Event handler "${handler}" found in output`);
    54→    }
    55→  }
    56→
    57→  // 3. javascript: URLs
    58→  if (/javascript\s*:/i.test(html)) {
    59→    violations.push(`"javascript:" URL scheme found in output`);
    60→  }
    61→
    62→  // 4. <iframe tag
    63→  if (/<iframe[\s>]/i.test(html)) {
    64→    violations.push(`<iframe> tag found in output (unescaped)`);
    65→  }
    66→
    67→  // 5. <img src=x pattern (unescaped)
    68→  if (/<img\s+src\s*=\s*x/i.test(html)) {
    69→    violations.push(`<img src=x> pattern found in output (unescaped)`);
    70→  }
    71→
    72→  // 6. <svg tag (unescaped)
    73→  if (/<svg[\s>]/i.test(html)) {
    74→    violations.push(`<svg> tag found in output (unescaped)`);
    75→  }
    76→
    77→  // 7. Check that dangerous characters in the payload are properly escaped
    78→  //    in the HTML output (only check for payloads that contain these chars)
    79→  if (payload.includes('<') && !html.includes('&lt;') && !html.includes('&gt;')) {
    80→    // If the payload has < but the output has neither &lt; nor the raw payload
    81→    // this could be a sign of incomplete escaping — but we already check above
    82→    // for specific patterns, so this is a secondary check
    83→  }
    84→
    85→  return { safe: violations.length === 0, violations };
    86→}
    87→
    88→/**
    89→ * More specific check: verify that the payload's dangerous chars
    90→ * are properly escaped in the output.
    91→ */
    92→function checkEscaping(html: string, payload: string): { properlyEscaped: boolean; details: string } {
    93→  // If the payload is present in raw form in the HTML, it's NOT escaped
    94→  if (html.includes(payload)) {
    95→    return { properlyEscaped: false, details: `Raw payload "${payload.slice(0, 50)}..." found unescaped in output` };
    96→  }
    97→
    98→  // Check specific character escaping
    99→  if (payload.includes('<') && !html.includes('&lt;') && payload.includes('>')) {
   100→    // The payload had angle brackets but there's no &lt; in the output
   101→    // This means it might have been stripped entirely rather than escaped
   102→    return { properlyEscaped: true, details: 'Payload not present; may have been stripped or filtered' };
   103→  }
   104→
   105→  return { properlyEscaped: true, details: 'Payload not found in raw form; appears properly escaped or absent' };
   106→}

   443→// ═══════════════════════════════════════════════════════════════════════
   444→describe('Fill-Blank — CRITICAL innerHTML DOM decoding vulnerability', () => {
   445→  it('data-answer is escaped in HTML source but will be decoded by DOM dataset API', () => {
   446→    const xssAnswer = '<script>window.__xss = true</script>';
   447→    const block = {
   448→      id: 'fb-critical-test',
   449→      title: 'Critical Test',
   450→      questions: [
   451→        { text: 'Fill in: ___', answer: xssAnswer, hint: 'hint' },
   452→      ],
   453→    };
   454→    const html = renderQuizBlock('fill-blank-game', block, noopRender)!;
   455→
   456→    // STEP 1: The HTML source should have the escaped version
   457→    expect(html, 'HTML source should contain &lt;script&gt; (escaped)').toContain('&lt;script&gt;');
   458→    expect(html, 'HTML source should NOT contain raw <script> tag').not.toMatch(/<script[\s>]/i);
   459→
   460→    // STEP 2: The data-answer attribute value is escaped in the HTML source
   461→    const dataAnswerMatch = html.match(/data-answer="([^"]*)"/);
   462→    expect(dataAnswerMatch, 'data-answer attribute should exist').not.toBeNull();
   463→    const attrValueInSource = dataAnswerMatch![1];
   464→    expect(attrValueInSource, 'data-answer in HTML source should be escaped').toContain('&lt;script&gt;');
   465→
   466→    // STEP 3: DOCUMENT THE VULNERABILITY
   467→    // The DOM automatically decodes HTML entities when reading data attributes.
   468→    // input.dataset.answer will return the DECODED string: <script>window.__xss = true</script>
   469→    // The runtime JS (scripts.ts line 479) does:
   470→    //   fb.innerHTML = '<span>...Jawaban: ' + input.dataset.answer + '</span>';
   471→    // This means the decoded <script> tag gets inserted into innerHTML,
   472→    // which the browser will NOT execute as a script tag (innerHTML script tags don't execute),
   473→    // BUT it WILL execute via other vectors like <img onerror>, <svg onload>, etc.
   474→
   475→    // Let's verify the DOM decoding behavior by simulating it:
   476→    // In a real browser, setting data-answer="&lt;script&gt;..." and reading
   477→    // via element.dataset.answer returns "<script>..."
   478→    // We simulate this by decoding the HTML entities:
   479→    const decodedViaDOM = attrValueInSource
   480→      .replace(/&amp;/g, '&')
