    19→// SAFE RICH TEXT — Sprint 6.3-B: Preserve safe HTML formatting in export
    20→// ═══════════════════════════════════════════════════════════════════════
    21→// In the canvas, DefBox and other blocks support <strong>, <em>, <br>, etc.
    22→// But the export pipeline used escapeHtml() which turned these into visible
    23→// &lt;strong&gt; text — destroying the teacher's formatting.
    24→//
    25→// safeRichText() preserves ONLY whitelisted safe tags and escapes
    26→// everything else — no XSS vector.
    27→// ═══════════════════════════════════════════════════════════════════════
    28→const SAFE_TAGS = new Set(['strong', 'b', 'em', 'i', 'u', 'br', 'p', 'ul', 'ol', 'li']);
    29→
    30→/**
    31→ * Preserve safe HTML formatting tags while escaping everything else.
    32→ * Whitelist: strong, b, em, i, u, br, p, ul, ol, li
    33→ * All other tags are escaped (e.g., <script> → &lt;script&gt;).
    34→ */
    35→function safeRichText(content: string): string {
    36→  if (!content) return '';
    37→  // Regex matches HTML tags (opening, closing, or self-closing)
    38→  return content.replace(/<\/(\w+)>|<(\w+)(\s[^>]*)?\/?>|<([^>]+)>/g, (match, closingTag, openTag, attrs, leftover) => {
    39→    // Closing tag: </strong>
    40→    if (closingTag) {
    41→      const tag = closingTag.toLowerCase();
    42→      if (SAFE_TAGS.has(tag)) return `</${tag}>`;
    43→      return escapeHtml(match);
    44→    }
    45→    // Opening/self-closing tag: <strong>, <br>, <br/>, <p class="x">
    46→    if (openTag) {
    47→      const tag = openTag.toLowerCase();
    48→      if (SAFE_TAGS.has(tag)) {
    49→        // For safe tags, allow them through but strip attributes (except <br>)
    50→        // to prevent event handlers like <strong onclick=...>
    51→        if (tag === 'br') return '<br>';
    52→        if (attrs && attrs.trim()) {
    53→          // Safe tag WITH attributes — strip attributes to prevent XSS
    54→          return `<${tag}>`;
    55→        }
    56→        return `<${tag}>`;
    57→      }
    58→      return escapeHtml(match);
    59→    }
    60→    // Leftover/malformed tag — escape it
    61→    return escapeHtml(match);
    62→  });
    63→}

    65→/**
    66→ * Render a content block. Returns null if the block type is not handled here.
    67→ */
    68→export function renderContentBlock(
    69→  type: string,
    70→  b: Record<string, unknown>,
    71→  renderBlock: RenderBlockFn,
    72→): string | null {
    73→  switch (type) {
    74→    case 'cover': return renderCover(b);
    75→    case 'petunjuk': return renderPetunjuk(b);
    76→    case 'tp': return renderTp(b);
    77→    case 'def-box': return renderDefBox(b);
    78→    case 'nc-grid': return renderNcGrid(b);
    79→    case 'nk-card': return renderNkCard(b);
    80→    case 'flashcard-set': return renderFlashcardSet(b);
    81→    case 'ftab': return renderFtab(b, renderBlock);
    82→    case 'materi-section': return renderMateriSection(b, renderBlock);
    83→    case 'tujuan-display': return renderTujuanDisplay(b);
    84→    case 'motivasi': return renderMotivasi(b);
    85→    case 'rangkuman': return renderRangkuman(b);
    86→    case 'diskusi': return renderDiskusi(b);
    87→    case 'hasil': return renderHasil(b);
    88→    case 'refleksi': return renderRefleksi(b);
    89→    case 'penutup': return renderPenutup(b);
    90→    case 'tabel-accord': return renderTabelAccord(b);
    91→    case 'gambar': return renderGambar(b);
    92→    case 'timeline': return renderTimeline(b);
    93→    case 'compare': return renderCompare(b);
    94→    case 'reveal': return renderReveal(b);
    95→    case 'materi-blok': return renderMateriBlok(b);
    96→    case 'tabel': return renderTabel(b);
    97→    case 'checklist': return renderChecklist(b);
    98→    case 'statistik': return renderStatistik(b);
    99→    case 'studi': return renderStudi(b);
   100→    case 'hero': return renderHero(b);
   101→    // 'kuis', 'alur', 'skenario' intentionally NOT handled here —
   102→    // they fall through to dedicated quiz-renderers.ts and navigation-renderers.ts
   103→    // which provide richer CSS-class-based output with interactivity.
   104→    default: return null;
   105→  }
   106→}

   108→/** Generic fallback renderer for unknown block types */
   109→export function renderGenericBlock(b: Record<string, unknown>): string {
   110→  const blockTitle = (b.title as string) || (b.type as string) || 'Block';
   111→  const content = (b.content as string) || '';
   112→  return `
   113→    <div class="block generic-block">
   114→      <div class="block-header">
   115→        <span class="block-icon">📦</span>
   116→        <h2>${escapeHtml(String(blockTitle))}</h2>
   117→      </div>
   118→      ${content ? `<p>${escapeHtml(content)}</p>` : '<p class="text-muted">Blok tidak didukung oleh export client-side.</p>'}
   119→    </div>`;
   120→}

   124→function renderCover(b: Record<string, unknown>): string {
   125→  const icon = b.icon as string || '📘';
   126→  const title = b.title as string || '';
   127→  const subtitle = b.subtitle as string || '';
   128→  const badges = (b.badges as Array<{ icon?: string; text: string; color: string }>) || [];
   129→  const accentColor = resolveColor(b.accentColor as string, '#fbbf24');
   130→  const bg = b.background as { color1?: string; color2?: string } | undefined;
   131→  const bgStyle = bg?.color1
   132→    ? `background: linear-gradient(135deg, ${bg.color1}, ${bg.color2 || bg.color1});`
   133→    : `background: linear-gradient(135deg, #ffffff, #f8fafc);`;
   134→
   135→  return `
   136→    <div class="block cover-block" style="${bgStyle}">
   137→      <div class="cover-glow" style="background: radial-gradient(circle at 50% 40%, ${accentColor}33, transparent 60%);"></div>
   138→      <div class="cover-icon">${icon}</div>
   139→      <h1 class="cover-title">${escapeHtml(title)}</h1>
   140→      <p class="cover-subtitle">${escapeHtml(subtitle)}</p>
   141→      <div class="cover-badges">
   142→        ${badges.map(bd => `<span class="badge" style="background:${resolveColor(bd.color, '#fbbf24')}22;color:${resolveColor(bd.color, '#fbbf24')};border:1px solid ${resolveColor(bd.color, '#fbbf24')}33;">${bd.icon || ''} ${escapeHtml(bd.text)}</span>`).join('')}
   143→      </div>
   144→    </div>`;
   145→}

   225→  const label = b.label as string || '';

   241→        <div class="nk-chars">
   242→          ${characteristics.map(ch => `<div class="nk-char"><strong>${escapeHtml(ch.label)}:</strong> ${escapeHtml(ch.value)}</div>`).join('')}

   278→function renderFtab(b: Record<string, unknown>, renderBlock: RenderBlockFn): string {
   279→  const tabs = (b.tabs as Array<{ icon: string; label: string; content: SchemaBlock[] }>) || [];
   280→  const tabId = `ftab-${Math.random().toString(36).slice(2, 8)}`;
   281→  return `
   282→    <div class="block ftab-block">
   283→      <div class="ftab-tabs" id="${tabId}">
   284→        ${tabs.map((t, i) => `
   285→          <button class="ftab-btn${i === 0 ? ' active' : ''}" onclick="switchFtab('${tabId}',${i})">${t.icon} ${escapeHtml(t.label)}</button>`).join('')}
   286→      </div>
   287→      ${tabs.map((t, i) => `
   288→        <div class="ftab-panel${i === 0 ? ' active' : ''}" data-ftab="${tabId}" data-idx="${i}">
   289→          ${t.content?.map(cb => renderBlock(cb)).join('') || ''}
   290→        </div>`).join('')}
   291→    </div>`;
   292→}

   294→function renderMateriSection(b: Record<string, unknown>, renderBlock: RenderBlockFn): string {
   295→  const title = b.title as string || 'Materi';
   296→  const subtitle = b.subtitle as string || '';
   297→  const icon = b.icon as string || '📖';
   298→  const accentColor = resolveColor(b.accentColor as string, '#a78bfa');
   299→  const variant = (b.variant as string) || 'A'; // Sprint 6.3-B: read variant
   300→  const content = (b.content as SchemaBlock[]) || [];
   301→  const tabs = b.tabs as Array<{ id: string; label: string; icon?: string; content: SchemaBlock[] }> | undefined;
   302→  const takeaways = (b.takeaways as string[]) || [];
   303→  const selfCheck = b.selfCheck as string | undefined;
   304→  const bsnpRequired = b.bsnpRequired as boolean;
   305→
   306→  // ── Sprint 6.3-B: Variant-aware wrapper styles ──
   307→  // Each variant gets a different CSS class + wrapper style so the export
   308→  // output doesn't silently fall back to Klasik when the teacher chose Majalah/Pill.
   309→  const variantClass = variant === 'B' ? 'materi-section-variant-majalah' : variant === 'C' ? 'materi-section-variant-pill' : '';
   310→  const variantStyle = variant === 'B'
   311→    ? 'border-left: 4px solid ' + accentColor + ';'
   312→    : variant === 'C'
   313→    ? 'border-left: 3px solid ' + accentColor + ';'
   314→    : 'border-left: 4px solid ' + accentColor + ';';
   315→
   316→  // ── Sprint 6.3-B: Tab content rendering ──
   317→  // When block.tabs is present, render ALL tab content (not just active tab).
   318→  // Each tab is rendered as a section with its label and content blocks.
   319→  const hasTabs = tabs && tabs.length > 0;
   320→  const tabContentHtml = hasTabs
   321→    ? (tabs as Array<{ id: string; label: string; icon?: string; content: SchemaBlock[] }>).map((tab, i) => `
   322→        <div class="materi-tab-section" style="margin-bottom: 12px;">
   323→          <div class="materi-tab-label" style="font-size: 11px; font-weight: 700; text-transform: uppercase; letter-spacing: 0.06em; color: ${accentColor}; margin-bottom: 6px; padding: 4px 10px; background: ${accentColor}0d; border-radius: 6px; display: inline-block;">${tab.icon ? tab.icon + ' ' : ''}${escapeHtml(tab.label)}</div>
   324→          <div class="materi-tab-content">
   325→            ${(tab.content || []).map(cb => renderBlock(cb)).join('')}
   326→          </div>
   327→        </div>`).join('')
   328→    : '';
   329→
   330→  // ── Variant B "Majalah": 2-column layout (content left, takeaways sidebar) ──
   331→  if (variant === 'B') {
   332→    return `
   333→    <div class="block materi-section-block ${variantClass}" style="${variantStyle}">
   334→      <div class="materi-section-header">
   335→        <span class="materi-section-icon">${icon}</span>
   336→        <div>
   337→          <h2>${escapeHtml(title)}</h2>
   338→          ${subtitle ? `<p class="materi-subtitle">${escapeHtml(subtitle)}</p>` : ''}
   339→        </div>
   340→        ${bsnpRequired ? '<span class="bsnp-badge">WAJIB BSNP</span>' : ''}
   341→      </div>
   342→      <div style="display: grid; grid-template-columns: 1fr 200px; gap: 16px; padding: 12px;">
   343→        <div class="materi-content">
   344→          ${hasTabs ? tabContentHtml : content.map(cb => renderBlock(cb)).join('')}
   345→        </div>
   346→        ${takeaways.length > 0 ? `
   347→        <div class="takeaways" style="background: ${accentColor}08; border: 1px solid ${accentColor}20; border-radius: 10px; padding: 10px;">
   348→          <h3 style="font-size: 10px; font-weight: 800; text-transform: uppercase; letter-spacing: 0.06em; color: ${accentColor}; margin-bottom: 8px;">⭐ Poin Penting</h3>
   349→          ${takeaways.map(t => `<div class="takeaway-item" style="font-size: 11px; color: #94a3b8; padding: 4px 0;">✓ ${escapeHtml(t)}</div>`).join('')}
   350→        </div>` : '<div></div>'}
   351→      </div>
   352→      ${selfCheck ? `
   353→        <div class="self-check" style="background: ${accentColor}08; border-left: 3px solid ${accentColor}; border-radius: 8px; padding: 10px 12px; margin: 0 12px 12px;">
   354→          <h3 style="font-size: 10px; font-weight: 800; color: ${accentColor};">🧠 Cek Pemahaman:</h3>
   355→          <p style="font-size: 12px; color: #f1f5f9;">${escapeHtml(selfCheck)}</p>
   356→        </div>` : ''}
   357→    </div>`;
   358→  }
   359→
   360→  // ── Variant C "Pill": Compact style with pill takeaway badges ──
   361→  if (variant === 'C') {
   362→    return `
   363→    <div class="block materi-section-block ${variantClass}" style="${variantStyle}">
   364→      <div class="materi-section-header" style="display: flex; align-items: center; gap: 8px; padding: 8px 12px;">
   365→        <span class="materi-section-icon" style="font-size: 14px;">${icon}</span>
   366→        <h2 style="font-size: 15px; font-weight: 700; color: #f1f5f9;">${escapeHtml(title)}</h2>
   367→        ${bsnpRequired ? '<span class="bsnp-badge" style="font-size: 9px;">WAJIB</span>' : ''}
   368→      </div>
   369→      <div class="materi-content" style="padding: 0 12px;">
   370→        ${hasTabs ? tabContentHtml : content.map(cb => renderBlock(cb)).join('')}
   371→      </div>
   372→      ${takeaways.length > 0 ? `
   373→      <div style="padding: 6px 12px;">
   374→        <div style="display: flex; flex-wrap: wrap; gap: 4px;">
   375→          ${takeaways.map(t => `<span style="display: inline-block; font-size: 10px; padding: 2px 8px; background: ${accentColor}15; color: ${accentColor}; border: 1px solid ${accentColor}30; border-radius: 12px;">✓ ${escapeHtml(t)}</span>`).join('')}
   376→        </div>
   377→      </div>` : ''}
   378→      ${selfCheck ? `
   379→        <div class="self-check" style="background: ${accentColor}08; border-left: 2px solid ${accentColor}; border-radius: 6px; padding: 6px 10px; margin: 6px 12px 10px;">
   380→          <p style="font-size: 11px; color: #f1f5f9;">🧠 ${escapeHtml(selfCheck)}</p>
   381→        </div>` : ''}
   382→    </div>`;
   383→  }
   384→
   385→  // ── Variant A "Klasik" (default): Original full section style ──
   386→  return `
   387→    <div class="block materi-section-block" style="border-left: 4px solid ${accentColor};">
   388→      <div class="materi-section-header">
   389→        <span class="materi-section-icon">${icon}</span>
   390→        <div>
   391→          <h2>${escapeHtml(title)}</h2>
   392→          ${subtitle ? `<p class="materi-subtitle">${escapeHtml(subtitle)}</p>` : ''}
   393→        </div>
   394→        ${bsnpRequired ? '<span class="bsnp-badge">WAJIB BSNP</span>' : ''}
   395→      </div>
   396→      <div class="materi-content">
   397→        ${hasTabs ? tabContentHtml : content.map(cb => renderBlock(cb)).join('')}
   398→      </div>
   399→      ${takeaways.length > 0 ? `
   400→        <div class="takeaways">
   401→          <h3>⭐ Poin Penting</h3>
   402→          ${takeaways.map(t => `<div class="takeaway-item">✓ ${escapeHtml(t)}</div>`).join('')}
   403→        </div>` : ''}
   404→      ${selfCheck ? `
   405→        <div class="self-check">
   406→          <h3>🧠 Apa yang sudah kamu pelajari?</h3>
   407→          <p>${escapeHtml(selfCheck)}</p>
   408→        </div>` : ''}
   409→    </div>`;
   410→}

   768→        </div>` : ''}
   769→    </div>`;

   785→function renderMateriBlok(b: Record<string, unknown>): string {
   786→  const tipe = b.tipe as string || 'teks';
   787→  const judul = b.judul as string || '';
   788→  const isi = b.isi as string || '';
   789→  const icon = b.icon as string || '';
   790→  const warna = b.warna as string || '';
   791→  const butir = b.butir as string[] || [];
   792→  const baris = b.baris as string[][] || [];
   793→  const langkah = b.langkah as Array<{ icon: string; judul: string; isi: string }> || [];
   794→  const kiri = b.kiri as { icon?: string; judul?: string; isi?: string } | undefined;
   795→  const kanan = b.kanan as { icon?: string; judul?: string; isi?: string } | undefined;
   796→  const karakter = b.karakter as string || '';
   797→  const situasi = b.situasi as string || '';
   798→  const pertanyaan = b.pertanyaan as string || '';
   799→  const infoboxStyle = b.infoboxStyle as string || 'info';
   800→  const items = b.items as Array<{ icon?: string; angka?: string; satuan?: string; label?: string; warna?: string }> || [];
   801→
   802→  // S2J.5: Default hex colors per safe subtipe, matching MateriBlokRenderer TIPE_META defaults.
   803→  // Used as fallback when block.warna is not set. resolveColor() maps token keys ('y','c',etc.)
   804→  // to actual hex colors, so warna='y' → '#fbbf24', warna=undefined → fallback hex.
   805→  const DEFAULT_ACCENT: Record<string, string> = {
   806→    teks: '#fbbf24',      // y (Kuning)
   807→    definisi: '#fbbf24',  // y
   808→    poin: '#3ecfcf',      // c (Cyan)
   809→    tabel: '#a78bfa',     // p (Ungu)
   810→    kutipan: '#34d399',   // g (Hijau)
   811→    gambar: '#3ecfcf',    // c
   812→    timeline: '#3ecfcf',  // c
   813→    highlight: '#fbbf24', // y
   814→    checklist: '#34d399', // g
   815→  };
   816→
   817→  // Sprint 6.3-B: accentColor takes priority over legacy warna, matching runtime MateriBlokRenderer
   818→  // Runtime uses: block.accentColor ?? block.warna ?? TIPE_META[tipe].color
   819→  const accentSource = (b.accentColor as string) || warna;
   820→  const c = resolveColor(accentSource, DEFAULT_ACCENT[tipe] || '#3ecfcf');
   821→
   822→  switch (tipe) {
   823→    case 'teks':
   824→      return `<div class="block materi-blok" style="border-left: 3px solid ${c}; padding: 12px; background: ${c}0d; border-radius: 10px;">${judul ? `<h3 style="color: ${c}; font-size: 13px;">${escapeHtml(judul)}</h3>` : ''}${isi ? `<p style="font-size: 11px; color: #94a3b8;">${escapeHtml(isi)}</p>` : ''}</div>`;
   825→    case 'definisi':
   826→      return `<div class="block materi-blok" style="border-left: 4px solid ${c}; padding: 12px; background: ${c}0d; border-radius: 10px;">${judul ? `<h3 style="color: ${c}; font-size: 13px;">📖 ${escapeHtml(judul)}</h3>` : ''}${isi ? `<p style="font-size: 11px; color: #94a3b8;">${escapeHtml(isi)}</p>` : ''}</div>`;
   827→    case 'poin':
   828→      return `<div class="block materi-blok" style="border-left: 3px solid ${c}; padding: 12px; background: ${c}0d; border-radius: 10px;">${judul ? `<h3 style="color: ${c}; font-size: 13px;">📋 ${escapeHtml(judul)}</h3>` : ''}<ul style="margin: 6px 0; padding-left: 16px;">${butir.map(b => `<li style="font-size: 11px; color: #94a3b8;">${escapeHtml(b)}</li>`).join('')}</ul></div>`;
   829→    case 'tabel':
   830→      return `<div class="block materi-blok" style="border-left: 3px solid ${c}; padding: 12px; background: ${c}0d; border-radius: 10px;">${judul ? `<h3 style="color: ${c}; font-size: 13px;">📊 ${escapeHtml(judul)}</h3>` : ''}${baris.length > 0 ? `<table style="width:100%;border-collapse:collapse;font-size:11px;"><tbody>${baris.map((r, ri) => `<tr>${r.map(cell => `<td style="padding:6px 8px;border:1px solid #334155;${ri === 0 ? `font-weight:700;background:${c}18;` : 'color:#94a3b8;'}">${escapeHtml(cell)}</td>`).join('')}</tr>`).join('')}</tbody></table>` : ''}</div>`;
   831→    case 'kutipan':
   832→      return `<div class="block materi-blok" style="border-left: 4px solid ${c}; padding: 16px; background: ${c}0d; border-radius: 10px; font-style: italic;">${isi ? `<p style="font-size: 13px; color: #f1f5f9;">"${escapeHtml(isi)}"</p>` : ''}${karakter ? `<p style="font-size: 11px; color: ${c}; margin-top: 6px;">— ${escapeHtml(karakter)}</p>` : ''}</div>`;
   833→    case 'highlight':
   834→      return `<div class="block materi-blok" style="border-left: 4px solid ${c}; padding: 12px; background: ${c}0d; border-radius: 10px;">${icon ? `<span style="font-size: 16px;">${icon}</span>` : ''}${judul ? `<h3 style="color: ${c}; font-size: 13px;">${escapeHtml(judul)}</h3>` : ''}${isi ? `<p style="font-size: 11px; color: #94a3b8;">${escapeHtml(isi)}</p>` : ''}</div>`;
   835→    case 'compare':
   836→      return renderCompare({ title: judul, kiri, kanan, accentColor: accentSource || 'c' });
   837→    case 'infobox': {
   838→      // S2J.5: infobox NOT in safe subtipes — color comes from infoboxStyle semantic meaning
   839→      const ic = infoboxStyle === 'warning' ? '#fbbf24' : infoboxStyle === 'tip' ? '#34d399' : '#3ecfcf';
   840→      return `<div class="block materi-blok" style="border-left: 4px solid ${ic}; padding: 12px; background: ${ic}0d; border-radius: 10px;">${judul ? `<h3 style="color: ${ic}; font-size: 13px;">ℹ️ ${escapeHtml(judul)}</h3>` : ''}${isi ? `<p style="font-size: 11px; color: #94a3b8;">${escapeHtml(isi)}</p>` : ''}</div>`;
   841→    }
   842→    case 'checklist':
   843→      return `<div class="block materi-blok" style="border-left: 3px solid ${c}; padding: 12px; background: ${c}0d; border-radius: 10px;">${judul ? `<h3 style="color: ${c}; font-size: 13px;">✅ ${escapeHtml(judul)}</h3>` : ''}${butir.map(b => `<div style="font-size: 11px; color: #94a3b8;"><input type="checkbox" disabled style="accent-color:${c};" /> ${escapeHtml(b)}</div>`).join('')}</div>`;
   844→    case 'statistik':
   845→      // S2J.5: statistik NOT in safe subtipes — each item has its own warna
   846→      return `<div class="block materi-blok" style="border-left: 3px solid #3ecfcf; padding: 12px; background: #3ecfcf0d; border-radius: 10px;">${judul ? `<h3 style="color: #3ecfcf; font-size: 13px;">📈 ${escapeHtml(judul)}</h3>` : ''}<div style="display: grid; grid-template-columns: repeat(auto-fit, minmax(80px, 1fr)); gap: 8px;">${items.map(it => { const sc = resolveColor(it.warna, '#3ecfcf'); return `<div style="text-align: center; padding: 8px; background: ${sc}1a; border-radius: 8px;"><div style="font-size: 18px; font-weight: 900; color: ${sc};">${escapeHtml(it.angka || '0')}</div>${it.satuan ? `<div style="font-size: 8px; color: #64748b;">${escapeHtml(it.satuan)}</div>` : ''}<div style="font-size: 9px; color: #94a3b8;">${escapeHtml(it.label || '')}</div></div>`; }).join('')}</div></div>`;
   847→    case 'studi':
   848→      // S2J.5: studi NOT in safe subtipes — multi-color semantic (red situasi, yellow pertanyaan, green pesan)
   849→      return `<div class="block materi-blok" style="border-left: 4px solid #fbbf24; padding: 12px; background: #fbbf240d; border-radius: 10px;">${judul ? `<h3 style="color: #fbbf24; font-size: 13px;">📑 ${escapeHtml(judul)}</h3>` : ''}${situasi ? `<p style="font-size: 11px; color: #94a3b8;">${escapeHtml(situasi)}</p>` : ''}${pertanyaan ? `<p style="font-size: 11px; color: #fbbf24; margin-top: 6px;">❓ ${escapeHtml(pertanyaan)}</p>` : ''}</div>`;
   850→    case 'timeline':
   851→      return renderTimeline({ title: judul, steps: langkah.map(s => ({ icon: s.icon, label: s.judul, description: s.isi, color: accentSource || 'c' })), accentColor: accentSource || 'c' });
   852→    case 'gambar':
   853→      return renderGambar({ title: judul, url: isi, caption: '', accentColor: accentSource || 'c' });
   854→    default:
   855→      return `<div class="block materi-blok" style="border-left: 3px solid #3ecfcf; padding: 12px;">${judul ? `<h3>${escapeHtml(judul)}</h3>` : ''}${isi ? `<p style="font-size: 11px; color: #94a3b8;">${escapeHtml(isi)}</p>` : ''}</div>`;
   856→  }
   857→}
