   467→        <div className="fixed top-3 left-3 z-50">
   468→          <Button
   469→            variant="ghost"
   470→            size="sm"
   471→            onClick={() => setActivePanel('dashboard')}
   472→            className="bg-silse-surface-container-lowest/95 border border-silse-outline-variant/60 shadow-md text-silse-on-surface-variant hover:text-silse-on-surface gap-1.5 backdrop-blur-md rounded-xl px-3 py-1.5 text-[12px] font-semibold"
   473→          >
   474→            <span className="material-symbols-outlined" style={{ fontSize: '16px' }}>arrow_back</span>
   475→            {isSederhana ? 'Beranda' : 'Dashboard'}
   476→          </Button>
   477→        </div>
   478→      )}
   479→
   480→      {/* ── Preview mode: Floating back button ──── */}
   481→      {isPreview && (
   482→        <div className="fixed top-3 left-3 z-50">
   483→          <Button
   484→            variant="ghost"
   485→            size="sm"
   486→            onClick={() => setActivePanel('dashboard')}
   487→            className="bg-silse-surface-container-lowest/95 border border-silse-outline-variant/60 shadow-md text-silse-on-surface-variant hover:text-silse-on-surface gap-1.5 backdrop-blur-md rounded-xl px-3 py-1.5 text-[12px] font-semibold"
   488→          >
   489→            <span className="material-symbols-outlined" style={{ fontSize: '16px' }}>arrow_back</span>
   490→            {isSederhana ? 'Beranda' : 'Dashboard'}
   491→          </Button>
   492→        </div>
   493→      )}
   494→
   495→      {/* ── Main Area ───────────────────────────────────────── */}
   496→      <div className="flex-1 flex flex-col min-w-0 min-h-0 overflow-hidden">
   497→        {/* ── Header Bar — SILSE v4 breadcrumb + actions ── */}
   498→        {!isCanva && !isPreview && (
   499→          <header className="h-12 flex-shrink-0 bg-silse-surface-container-lowest border-b border-silse-outline-variant/60 flex items-center gap-3 px-5">
   500→            <div className="flex items-center gap-2">
   501→              <span className="material-symbols-outlined text-silse-on-surface-variant" style={{ fontSize: '16px' }}>{
   502→                activePanel === 'dashboard' ? 'space_dashboard' :
   503→                activePanel === 'dokumen' ? 'edit_note' :
   504→                activePanel === 'konten' ? 'folder_open' :
   505→                activePanel === 'autogen' ? 'auto_awesome' :
   506→                activePanel === 'projects' ? 'folder_open' :
   507→                activePanel === 'import' ? 'swap_horiz' :
   508→                activePanel === 'versions' ? 'schedule' :
   509→                'space_dashboard'
   510→              }</span>
   511→              <span className="text-[13px] font-semibold text-silse-on-surface">
   512→                {panelTitles[activePanel]}
   513→              </span>
   514→              <span className="text-silse-outline-variant mx-0.5">/</span>
   515→              <span className="text-[12px] text-silse-on-surface-variant">{meta.judulPertemuan || 'Proyek Baru'}</span>
   516→            </div>
   517→
   518→            {/* Dirty indicator */}
   519→            <div className="flex items-center gap-1.5">
   520→              <div
   521→                className={`w-2 h-2 rounded-full flex-shrink-0 transition-opacity duration-300 ${
   522→                  dirty ? 'bg-silse-primary pulse-dot opacity-100' : 'bg-silse-primary/50 opacity-0'
   523→                }`}
   524→                title={dirty ? 'Perubahan belum disimpan' : 'Tersimpan'}
   525→              />
   526→              {dirty && (
   527→                <span className="text-[9px] text-silse-on-surface-variant font-medium">Belum simpan</span>
   528→              )}
   529→            </div>
   530→
   531→            {/* Workflow step indicator */}
   532→            <WorkflowStepIndicator />
   533→
   534→            <div className="ml-auto flex items-center gap-2">
   535→              <Button
   536→                variant="outline"
   537→                size="sm"
   538→                onClick={() => setActivePanel('preview')}
   539→                className="text-silse-primary border-silse-outline-variant/60 hover:bg-silse-primary/5 hover:text-silse-primary rounded-xl text-[12px] font-semibold gap-1.5"
   540→              >
   541→                <span className="material-symbols-outlined" style={{ fontSize: '14px' }}>visibility</span>
   542→                {isSederhana ? 'Pratinjau' : 'Preview'}
   543→              </Button>
   544→              <Button
   545→                variant="ghost"
   546→                size="sm"
   547→                onClick={() => setActivePanel('canva')}
   548→                className="text-silse-on-surface-variant hover:text-silse-on-surface hover:bg-silse-surface-container-high/60 rounded-xl text-[12px] font-semibold gap-1.5"
   549→              >
   550→                <span className="material-symbols-outlined" style={{ fontSize: '14px' }}>palette</span>
   551→                {isSederhana ? 'Desain' : 'Canva'}
   552→              </Button>
   553→              <Button
   554→                size="sm"
   555→                onClick={saveAll}
   556→                disabled={saving}
   557→                className="bg-silse-primary text-silse-on-primary hover:bg-silse-primary/90 disabled:opacity-50 font-semibold rounded-xl text-[12px] gap-1.5"
   558→              >
   559→                <span className="material-symbols-outlined" style={{ fontSize: '14px', animation: saving ? 'spin 1s linear infinite' : 'none' }}>save</span>
   560→                {saving ? '...' : 'Simpan'}
   561→              </Button>
   562→            </div>
   563→          </header>
   564→        )}
   565→
   566→        {/* ── Content ──────────────────────────────────────── */}
   567→        <div
   568→          key={activePanel}
   569→          role="main"
   570→          className={`flex-1 flex flex-col min-h-0 anim-enter-fade ${
   571→            isCanva || isPreview ? 'overflow-hidden' : 'overflow-y-auto bg-silse-surface-bright'
   572→          }`}
   573→        >
   574→          {renderPanel()}
   575→        </div>
   576→      </div>
   577→
   578→      {/* ── Template Wizard Modal ── */}
   579→      <TemplateWizard open={wizardOpen} onOpenChange={setWizardOpen} />
   580→
   581→      {/* ── Dev Performance Monitor (only in development) ── */}
   582→      <PerformanceMonitor />
   583→
   584→      {/* ── Unified Recovery Dialog ─────────────────────────── */}
   585→      <RecoveryDialog />
   586→
   587→      {/* ── Guided Tour Overlay ────────────────────────────── */}
   588→      {showTour && activePanel === 'dashboard' && (
   589→        <div className="fixed inset-0 z-50 flex items-center justify-center">
   590→          <div className="absolute inset-0 bg-silse-on-surface/40 backdrop-blur-sm" />
   591→          <div className="relative z-10 w-full max-w-sm mx-4 page-transition">
   592→            <div className="bg-silse-surface-container-lowest border border-silse-outline-variant/50 rounded-2xl shadow-2xl overflow-hidden">
   593→              <div className="bg-silse-primary-container/10 px-5 pt-5 pb-3">
   594→                <div className="flex items-center gap-3">
   595→                  <div className="w-10 h-10 rounded-lg bg-silse-primary-container/20 flex items-center justify-center">
   596→                    <span className="material-symbols-outlined text-silse-primary" style={{ fontSize: '20px' }}>location_on</span>
   597→                  </div>
   598→                  <div>
   599→                    <div className="text-xs font-medium text-silse-primary/70">
   600→                      Langkah {tourStep + 1} dari {TOUR_STEPS.length}
   601→                    </div>
   602→                    <h3 className="text-base font-bold text-silse-on-surface">
   603→                      {TOUR_STEPS[tourStep]!.title}
   604→                    </h3>
   605→                  </div>
   606→                </div>
   607→              </div>
   608→              <div className="px-5 py-4">
   609→                <p className="text-sm text-silse-on-surface-variant leading-relaxed">
   610→                  {TOUR_STEPS[tourStep]!.desc}
   611→                </p>
   612→              </div>
   613→              <div className="px-5 pb-2 flex justify-center gap-1.5">
   614→                {TOUR_STEPS.map((_, i) => (
   615→                  <span
   616→                    key={i}
   617→                    className={`block h-1.5 rounded-full transition-all duration-300 ${
   618→                      i === tourStep
   619→                        ? 'w-5 bg-silse-primary'
   620→                        : 'w-1.5 bg-silse-surface-container-high'
   621→                    }`}
   622→                  />
   623→                ))}
   624→              </div>
   625→              <div className="px-5 pb-5 pt-3 flex items-center gap-3">
   626→                <Button
   627→                  variant="ghost"
   628→                  onClick={dismissTour}
   629→                  className="px-4 py-2 text-xs font-medium"
   630→                >
   631→                  Lewati
   632→                </Button>
   633→                <Button
   634→                  onClick={nextTourStep}
   635→                  className="flex-1 bg-silse-primary text-silse-on-primary hover:bg-silse-primary/90 shadow-sm hover:shadow-md hover:-translate-y-px"
   636→                >
   637→                  {tourStep < TOUR_STEPS.length - 1 ? 'Berikutnya →' : 'Mulai ✨'}
   638→                </Button>
   639→              </div>
   640→            </div>
   641→          </div>
   642→        </div>
   643→      )}
   644→    </div>
   645→  );
   646→}
