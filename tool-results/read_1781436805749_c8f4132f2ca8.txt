    16→    16→// ── Background-image-aware styling helpers ────────────────────────
    17→    17→// When a background image is active, quiz options/feedback need more
    18→    18→// opaque backgrounds and visible borders so they remain readable
    19→    19→// on top of the image. When no bg image, original styles preserved.
    20→    20→
    21→    21→function kuisOptionBg(tokens: TokenResolver, isAnswered: boolean, isCorrect: boolean, isPicked: boolean): string {
    22→    22→  if (!tokens.hasBackgroundImage()) {
    23→    23→    if (!isAnswered) return tokens.subtleBg(0.04);
    24→    24→    if (isCorrect) return tokens.accentBg('g', 0.08);
    25→    25→    if (isPicked) return tokens.accentBg('r', 0.08);
    26→    26→    return tokens.subtleBg(0.04);
    27→    27→  }
    28→    28→  const isDark = tokens.isDark();
    29→    29→  if (!isAnswered) return isDark ? 'rgba(15,23,42,0.78)' : 'rgba(255,255,255,0.86)';
    30→    30→  if (isCorrect) {
    31→    31→    const tint = tokens.accentBg('g', 0.16);
    32→    32→    return `linear-gradient(${tint}, ${tint}), ${isDark ? 'rgba(15,23,42,0.78)' : 'rgba(255,255,255,0.86)'}`;
    33→    33→  }
    34→    34→  if (isPicked) {
    35→    35→    const tint = tokens.accentBg('r', 0.16);
    36→    36→    return `linear-gradient(${tint}, ${tint}), ${isDark ? 'rgba(15,23,42,0.78)' : 'rgba(255,255,255,0.86)'}`;
    37→    37→  }
    38→    38→  return isDark ? 'rgba(15,23,42,0.68)' : 'rgba(255,255,255,0.76)';
    39→    39→}
    40→    40→
    41→    41→function kuisOptionBorder(tokens: TokenResolver, isAnswered: boolean, isCorrect: boolean, isPicked: boolean): string {
    42→    42→  if (!tokens.hasBackgroundImage()) {
    43→    43→    if (!isAnswered) return tokens.subtleBorder(0.1);
    44→    44→    if (isCorrect) return tokens.color('g');
    45→    45→    if (isPicked) return tokens.color('r');
    46→    46→    return tokens.subtleBorder(0.08);
    47→    47→  }
    48→    48→  const isDark = tokens.isDark();
    49→    49→  if (!isAnswered) return isDark ? 'rgba(255,255,255,0.22)' : 'rgba(0,0,0,0.14)';
    50→    50→  if (isCorrect) return tokens.color('g');
    51→    51→  if (isPicked) return tokens.color('r');
    52→    52→  return isDark ? 'rgba(255,255,255,0.12)' : 'rgba(0,0,0,0.08)';
    53→    53→}
    54→    54→
    55→    55→function kuisFeedbackBg(tokens: TokenResolver, isCorrect: boolean): string {
    56→    56→  if (!tokens.hasBackgroundImage()) return tokens.accentBg(isCorrect ? 'g' : 'r', 0.06);
    57→    57→  const tint = tokens.accentBg(isCorrect ? 'g' : 'r', 0.16);
    58→    58→  const isDark = tokens.isDark();
    59→    59→  return `linear-gradient(${tint}, ${tint}), ${isDark ? 'rgba(15,23,42,0.82)' : 'rgba(255,255,255,0.88)'}`;
    60→    60→}

   353→   353→export const KuisRenderer = React.memo(function KuisRenderer({ block, tokens, interactive, isCompact, isEditing, pageIndex, compression }: {
   354→   354→  block: KuisBlock; tokens: TokenResolver; interactive: boolean; isCompact: boolean; isEditing?: boolean; pageIndex?: number; compression?: import('../../layout/CompressionEngine').CompressionDecision;
   355→   355→}) {
   356→   356→  const [current, setCurrent] = React.useState(0);
   357→   357→  const [answers, setAnswers] = React.useState<Record<number, number>>({});
   358→   358→  const [showExplanation, setShowExplanation] = React.useState(false);
   359→   359→
   360→   360→  // ── Educational design tokens ──────────────────────────────
   361→   361→  const edu = tokens.edu('kuis', isCompact);
   362→   362→
   363→   363→  // ── Variant state ───────────────────────────────────────────
   364→   364→  // Sprint 2K.4: Persist variant changes to schema (same pattern as other renderers)
   365→   365→  const updateSchemaBlock = useCanvaStore((s) => s.updateSchemaBlock);
   366→   366→  const [currentVariant, setCurrentVariant] = useState<'A' | 'B' | 'C'>(
   367→   367→    (block.variant as 'A' | 'B' | 'C') || 'A'
   368→   368→  );
   369→   369→  const variant = currentVariant;
   370→   370→  const handleVariantChange = useCallback((v: 'A' | 'B' | 'C') => {
   371→   371→    setCurrentVariant(v);
   372→   372→    if (block.id) updateSchemaBlock(block.id, { variant: v });
   373→   373→  }, [block.id, updateSchemaBlock]);
   374→   374→
   375→   375→  // ── Compression-aware rendering (step-reveal strategy) ──────
   376→   376→  // Kuis naturally shows one question at a time (step-reveal).
   377→   377→  // When compressed, we switch to the most compact variant (C "Ringkas")
   378→   378→  // and hide decorative elements (streak badge, progress aurora).
   379→   379→  const questions = block.questions || [];
   380→   380→  const { isCompressed } = useBlockCompression({
   381→   381→    compression,
   382→   382→    totalItems: questions.length,
   383→   383→  });
   384→   384→  const effectiveVariant = isCompressed ? 'C' as const : variant;
   385→   385→
   386→   386→  // ── Replay watcher: reset all state when replayGeneration bumps ──
   387→   387→  const replayGeneration = useInteractiveStore(s => s.replayGeneration);
   388→   388→  React.useEffect(() => {
   389→   389→    setCurrent(0);
   390→   390→    setAnswers({});
   391→   391→    setShowExplanation(false);
   392→   392→  }, [replayGeneration]);
   393→   393→
   394→   394→  const q = questions[current];
   395→   395→  const totalAnswered = React.useMemo(
   396→   396→    () => Object.keys(answers).length,
   397→   397→    [answers],
   398→   398→  );
   399→   399→  const totalCorrect = React.useMemo(
   400→   400→    () => Object.entries(answers).filter(([idx, ans]) => questions[Number(idx)]?.ans === ans).length,
   401→   401→    [answers, questions],
   402→   402→  );
   403→   403→  const isCompleted = totalAnswered >= questions.length && questions.length > 0;
   404→   404→
   405→   405→  // ── Streak calculation ──────────────────────────────────────
   406→   406→  const currentStreak = React.useMemo(() => {
   407→   407→    let s = 0;
   408→   408→    for (let i = current - 1; i >= 0; i--) {
   409→   409→      if (answers[i] !== undefined && questions[i]?.ans === answers[i]) s++;
   410→   410→      else break;
   411→   411→    }
   412→   412→    return s;
   413→   413→  }, [answers, current, questions]);
   414→   414→
