    59→// ── Schema Strip Helper ───────────────────────────────────────
    60→// Removes derived runtime fields that might have been written to
    61→// schema blocks before the purity enforcement was in place.
    62→// This is a belt-and-suspenders strip before JSON serialization.
    63→function stripRuntimeFieldsFromPages(pages: CanvaPage[]): CanvaPage[] {
    64→  return pages.map(page => {
    65→    if (!page.schema) return page;
    66→    return {
    67→      ...page,
    68→      schema: {
    69→        ...page.schema,
    70→        blocks: page.schema.blocks.map(block => {
    71→          // Strip _compressedHeight from compression hints (legacy data)
    72→          // Use explicit type cast to preserve CompressionHints type after delete
    73→          if (block.compression && '_compressedHeight' in block.compression) {
    74→            const cleanCompression = { ...block.compression };
    75→            delete (cleanCompression as Record<string, unknown>)._compressedHeight;
    76→            return { ...block, compression: cleanCompression };
    77→          }
    78→          return block;
    79→        }),
    80→      },
    81→    };
    82→  });
    83→}

    85→export const createPersistenceSlice: StateCreator<CanvaState, [], [], PersistenceSlice> = (set, get) => ({
    86→  // ── Persistence ──────────────────────────────────────────────
    87→  saveToStorage: () => {
    88→    try {
    89→      const { pages, ratioId } = get();
    90→
    91→      // ── Strip derived runtime fields before persistence ──
    92→      // Belt-and-suspenders: remove any _compressedHeight that might
    93→      // exist from older transaction writes (now moved to runtime cache).
    94→      const cleanPages = stripRuntimeFieldsFromPages(pages);
    95→
    96→      // ── Purity Guard: Ensure no runtime state leaks into persisted data ──
    97→      // In dev mode, throws if any schema contains runtime state fields.
    98→      // In production, logs the violation but continues saving.
    99→      // WRAPPED: Skip purity check if it throws (e.g., stack overflow from deep schema)
   100→      try {
   101→        for (const page of cleanPages) {
   102→          if (page.schema) {
   103→            assertDocumentPurity(page.schema, `saveToStorage page=${page.id}`);
   104→          }
   105→        }
   106→      } catch (purityErr) {
   107→        logger.warn('CanvaStore', 'Purity check skipped: ' + String(purityErr));
   108→      }
   109→
   110→      // Safe JSON.stringify with circular reference detection
   111→      // Uses a path-based depth limiter to prevent stack overflow
   112→      // on deeply nested or corrupted schema data.
   113→      const seen = new WeakSet();
   114→      const safeStringify = (obj: unknown): string => {
   115→        return JSON.stringify(obj, function(this: unknown, _key: string, value: unknown): unknown {
   116→          if (typeof value === 'object' && value !== null) {
   117→            if (seen.has(value)) return '[Circular]';
   118→            seen.add(value);
   119→          }
   120→          return value;
   121→        });
   122→      };
   123→
   124→      localStorage.setItem(CANVA_STORAGE_KEY, safeStringify({
   125→        pages: cleanPages,
   126→        ratioId,
   127→        _lastSavedAt: Date.now(),
   128→        _migrationVersion: STORAGE_MIGRATION_VERSION,
   129→        _schemaHash: computePagesHash(cleanPages),
   130→      }));
   131→      // Sprint 7.1: Do NOT set _saveStatus to 'saved' here.
   132→      // _saveStatus is now managed by the revision-based state machine.
   133→      // saveToStorage() is a LOCAL backup — durable save is the DB path.
   134→      set({ _lastSavedAt: Date.now(), _pagesHashAtSave: computePagesHash(cleanPages) });
   135→    } catch (err) {
   136→      // Storage full, unavailable, or stack overflow from corrupted data
   137→      logger.warn('CanvaStore', 'Failed to save to localStorage: ' + String(err));
   138→      // If stack overflow, clear localStorage to break the cycle
   139→      if (err instanceof RangeError) {
   140→        logger.warn('CanvaStore', 'Stack overflow detected — clearing corrupted localStorage data');
   141→        try { localStorage.removeItem(CANVA_STORAGE_KEY); } catch {}
   142→      }
   143→      set({ _saveStatus: 'error' });
   144→    }
   145→  },
   146→
   147→  loadFromStorage: () => {
   148→    try {
   149→      // ── FASE 6: Crash Recovery ──────────────────────────────
   150→      // Check if there's a crash recovery checkpoint (from a previous
   151→      // session that crashed during a dangerous operation).
   152→      // If so, prefer the checkpoint over the last auto-save.
   153→      const crashMeta = hasCrashRecovery();
   154→      let raw: string | null = null;
   155→
   156→      if (crashMeta) {
   157→        const crashData = loadCrashRecovery();
   158→        if (crashData) {
   159→          // Use crash recovery data — it's more recent than auto-save
   160→          raw = JSON.stringify({ pages: crashData.pages, ratioId: crashData.ratioId });
   161→          if (process.env.NODE_ENV === 'development') console.log(
   162→            `[Recovery] Found crash checkpoint from ${new Date(crashMeta.timestamp).toLocaleTimeString()} ` +
   163→            `(reason: ${crashMeta.reason}, ${crashMeta.pageCount} pages)`
   164→          );
   165→          clearCrashRecovery();
   166→        }
   167→      }
   168→
   169→      // Fall back to normal storage if no crash recovery
   170→      if (!raw) {
   171→        raw = localStorage.getItem(CANVA_STORAGE_KEY);
   172→      }
   173→      if (!raw) return false;
   174→
   175→      // Clear runtime caches when loading new project data.
   176→      clearCompressedHeightCache();
   177→      clearMeasurementCache();
   178→
   179→      const data = JSON.parse(raw);
   180→      if (data.pages && Array.isArray(data.pages)) {
   181→        // Ensure all pages have new fields (backward compat)
   182→        // FASE 1: Auto-migrate legacy pages on load — ensurePageSchema()
   183→        // proactively migrates all template pages so page.schema is
   184→        // populated BEFORE first render. This means:
   185→        //   - Legacy pages (no schema field) → migrated via TemplateAdapter
   186→        //   - Pages with schemaScreen in templateData → promoted to page.schema
   187→        //   - Already-native schema pages → preserved as-is
   188→        // After this, every template page has page.schema set.
   189→        // The TemplateAdapter is never called again for these pages.
   190→        //
   191→        // DUAL-RENDER FIX: migrateAllPages() also clears elements[]
   192→        // for any page that has schema. Schema-driven pages must NOT
   193→        // have elements[] populated, otherwise dual-render occurs.
   194→        // Check if overlayElements migration has already been applied
   195→        const needsOverlayMigration = !data._migrationVersion || data._migrationVersion < 1;
   196→
   197→        const rawPages = data.pages.map((p: CanvaPage) => {
   198→          // Basic field migration (backward compat)
   199→          const migrated: CanvaPage = {
   200→            ...p,
   201→            templateType: p.templateType || 'custom',
   202→            colorPalette: p.colorPalette || null,
   203→            navConfig: {
   204→              ...DEFAULT_NAV_CONFIG,
   205→              ...(p.navConfig || {}),
   206→              navbarStyle: p.navConfig?.navbarStyle || DEFAULT_NAV_CONFIG.navbarStyle,
   207→            },
   208→            templateData: p.templateData || {},
   209→            elements: [
   210→              ...(p.elements || []).map((el: CanvaElement) => ({
   211→                ...el,
   212→                opacity: el.opacity ?? 100,
   213→                hidden: el.hidden ?? false,
   214→              })),
   215→              // v4 (migration v1): Merge overlayElements into elements[] once.
   216→              // After the first save with _migrationVersion >= 1,
   217→              // overlayElements is always [] and this is a no-op.
   218→              ...(needsOverlayMigration ? (p.overlayElements || []).map((el: CanvaElement) => ({
   219→                ...el,
   220→                opacity: el.opacity ?? 100,
   221→                hidden: el.hidden ?? false,
   222→              })) : []),
   223→            ],
   224→            overlayElements: [], // Always cleared — legacy field
   225→            schema: p.schema || undefined,
   226→          };
   227→          return migrated;
   228→        });
   229→
   230→        // Apply schema migration + clear elements[] for schema pages
   231→        const pages = migrateAllPages(rawPages);
   232→
   233→        // D-P0B.3: Apply schema version migrations (v0→v1, etc.) and USE the result.
   234→        // Previously only migratedCount was captured but the migrated pages array was discarded.
   235→        const { pages: migratedPages, migratedCount } = migrateAllSchemas(pages);
   236→        if (migratedCount > 0 && process.env.NODE_ENV === 'development') {
   237→          console.log(`[Persistence] Migrated ${migratedCount} page schemas to latest version`);
   238→        }
   239→
   240→        // ── Strip derived runtime fields from legacy data ──
   241→        // Older versions may have _compressedHeight in compression hints.
   242→        // Remove it before the purity check so we don't throw on old data.
   243→        const cleanPages = stripRuntimeFieldsFromPages(migratedPages);
   244→
   245→        // ── Purity Guard: Check loaded data for runtime state leakage ──
   246→        // WRAPPED: Skip purity check if it throws (e.g., stack overflow from deep/corrupted schema)
   247→        try {
   248→          for (const page of cleanPages) {
   249→            if (page.schema) {
   250→              assertDocumentPurity(page.schema, `loadFromStorage page=${page.id}`);
   251→            }
   252→          }
   253→        } catch (purityErr) {
   254→          logger.warn('CanvaStore', 'Purity check on load failed (data may be corrupted): ' + String(purityErr));
   255→        }
   256→
   257→        // ── FASE 6: Proactive integrity validation ──────────────────
   258→        // Validate all page schemas after migration and BEFORE setting state.
   259→        // This catches corruption early and auto-repairs if possible.
   260→        try {
   261→          const validationResult = validateAndRepairPages(cleanPages, { autoRepair: true });
   262→          if (validationResult.repairedPages > 0) {
   263→            logger.warn('Recovery', `Proactive repair: ${validationResult.repairedPages}/${validationResult.totalPages} pages repaired`);
   264→            // Set safe mode flag — some data was corrupted
   265→            try { sessionStorage.setItem('silse_safe_mode', '1'); } catch {}
   266→          }
   267→          if (validationResult.corruptedPages > 0 && validationResult.repairedPages === 0) {
   268→            logger.error('Recovery', `Unrecoverable corruption: ${validationResult.corruptedPages} pages`);
   269→            try { sessionStorage.setItem('silse_safe_mode', '1'); } catch {}
   270→          }
   271→          set({ _lastIntegrityResult: validationResult });
   272→        } catch (validationErr) {
   273→          logger.warn('Recovery', 'Proactive validation failed: ' + String(validationErr));
   274→        }
   275→
   276→        // ── FASE 6: Hash verification on load ──────────────────
   277→        // Verify stored hash matches computed hash (detects in-transit corruption)
   278→        if (data._schemaHash) {
   279→          const currentHash = computePagesHash(cleanPages);
   280→          if (currentHash !== data._schemaHash) {
   281→            logger.warn('Recovery', 'Schema hash mismatch on load — data may have been corrupted');
   282→            try { sessionStorage.setItem('silse_safe_mode', '1'); } catch {}
   283→          }
   284→        }
   285→
   286→        // ── P0-6 Fix: Hydration suppression with depth counter ──
   287→        // Suppress markDirty() during load to prevent false "unsaved" state.
   288→        //
   289→        // ORDERING: resetOnLoad() is called FIRST (resets revision counters),
   290→        // then startHydration() increments the depth counter. This ensures
   291→        // the depth counter is > 0 during the set() call that follows,
   292→        // which triggers store subscriptions that would otherwise mark dirty.
   293→        //
   294→        // resetOnLoad() does NOT touch _hydrationDepth (by design), so
   295→        // it's safe to call before startHydration().
   296→        useDirtyStore.getState().resetOnLoad();
   297→        useAuthoringStore.setState({ dirty: false });
   298→
   299→        // Start hydration AFTER resetOnLoad — depth counter increments,
   300→        // ensuring markDirty() is suppressed during the set() call below.
   301→        useDirtyStore.getState().startHydration();
   302→
   303→        // Migrate legacy leftTab names
   304→        let leftTab: LeftTab = 'pages';
   305→        if (data.leftTab) {
   306→          leftTab = TAB_MIGRATION[data.leftTab] || 'pages';
   307→        }
   308→        set({
   309→          pages: cleanPages,
   310→          ratioId: data.ratioId || '16:9',
   311→          currentPageIndex: 0,
   312→          kontenTabRequest: null, // Phase 3: reset ephemeral nav
   313→          kontenPanelRequest: false,
   314→          panelRequest: null,
   315→          selectedElId: null,
   316→          selectedElIds: [], // Phase 4: Reset multi-select on load
   317→          selectedBlockIds: [], // Reset block multi-select on load
   318→          leftPanelOpen: true,
   319→          rightPanelOpen: true,
   320→          leftTab,
   321→        });
   322→
   323→        // Sprint 7.2A-7: End hydration after set() completes
   324→        useDirtyStore.getState().endHydration();
   325→        return true;
   326→      }
   327→      return false;
   328→    } catch {
   329→      // Sprint 7.2A-7: End hydration even on load failure to prevent stuck state
   330→      try {
   331→        useDirtyStore.getState().endHydration();
   332→      } catch { /* best effort */ }
   333→
   334→      // FASE 6: Try safe boot before giving up
   335→      try {
   336→        const rawForRecovery = localStorage.getItem(CANVA_STORAGE_KEY);
   337→        const bootResult = safeBootFromStorage(rawForRecovery);
   338→        if (bootResult.booted && bootResult.repairs.length > 0) {
   339→          logger.warn('Recovery', 'Safe boot repairs: ' + bootResult.repairs.join(', '));
   340→          // Try to load the repaired data
   341→          const repairedData = JSON.parse(rawForRecovery!);
   342→          // Apply repairs to pages
   343→          const repairedPages = (repairedData.pages || []).map((p: Record<string, unknown>, _i: number) => {
   344→            if (bootResult.safeMode && p.schema) {
   345→              const repairResult = repairSchema(p.schema as import('@/core/schema/types').ScreenSchema);
   346→              return { ...p, schema: repairResult.schema };
   347→            }
   348→            return p;
   349→          });
   350→          // Set safe mode flag if data was repaired
   351→          if (bootResult.safeMode) {
   352→            try { sessionStorage.setItem('silse_safe_mode', '1'); } catch {}
   353→          }
   354→          // Re-save the repaired data
   355→          localStorage.setItem(CANVA_STORAGE_KEY, JSON.stringify({
   356→            ...repairedData,
   357→            pages: repairedPages,
   358→            _lastSavedAt: Date.now(),
   359→            _migrationVersion: 1,
   360→          }));
   361→          // Try loading again
   362→          return get().loadFromStorage();
   363→        }
   364→      } catch (recoveryErr) {
   365→        logger.warn('Recovery', 'Safe boot failed: ' + String(recoveryErr));
   366→      }
   367→      try { localStorage.removeItem(CANVA_STORAGE_KEY); } catch {}
   368→      return false;
   369→    }
   370→  },
   371→
   372→  // ── Load from Database ──────────────────────────────────────────
   373→  // Patch-3: TRANSACTIONAL loadFromDB — parse/validate ALL payloads
   374→  // BEFORE any store mutation. This prevents cross-project contamination
   375→  // when Project B's data is corrupted: if parsing fails, Project A's
   376→  // state remains untouched.
   377→  //
   378→  // Key changes from Patch-2:
   379→  //   1. resetOnLoad() REMOVED — Project Manager handles this only after success
   380→  //   2. authoring dirty NOT cleared until after successful parse
   381→  //   3. pages must be an array — null/undefined/other fails closed (P0-2)
   382→  //   4. authoringData parsed and validated before any mutation (P0-3)
   383→  //   5. All parsing is PURE (no store side effects)
   384→  //   6. Hydration + commit only happen after ALL parsing succeeds
   385→  //
   386→  // Returns true if ALL data was parsed and committed successfully.
   387→  // Returns false if any parsing failed — NO stores are mutated.
   388→  loadFromDB: (data: DBProjectData): boolean => {
   389→    try {
   390→      // ══════════════════════════════════════════════════════════
   391→      // PHASE 1: VALIDATE — Fail closed on invalid payload shape
   392→      // ══════════════════════════════════════════════════════════
   393→      // P0-2 Fix: pages MUST be an array. null, undefined, or any
   394→      // other type is a hard failure — we never mutate stores.
   395→      // Empty array [] is valid (blank project).
   396→      if (!Array.isArray(data.pages)) {
   397→        throw new Error('Invalid project pages payload: expected array, got ' + typeof data.pages);
   398→      }
   399→
   400→      // ══════════════════════════════════════════════════════════
   401→      // PHASE 2: PURE PREPARATION — Parse everything, touch nothing
   402→      // ══════════════════════════════════════════════════════════
   403→      // All JSON.parse, migrations, and transformations happen here.
   404→      // If ANY step throws, we catch it and return false WITHOUT
   405→      // having mutated any store. Project A stays intact.
   406→
   407→      // Parse and migrate canva pages
   408→      const rawPages: CanvaPage[] = data.pages.map((p) => {
   409→        const schema = p.schemaData ? JSON.parse(p.schemaData) : undefined;
   410→        const parsedNavConfig = p.navConfig ? JSON.parse(p.navConfig) : {};
   411→        const navConfig: NavConfig = {
   412→          ...DEFAULT_NAV_CONFIG,
   413→          ...parsedNavConfig,
   414→          navbarStyle: parsedNavConfig.navbarStyle || DEFAULT_NAV_CONFIG.navbarStyle,
   415→        };
   416→        const templateData = p.templateData ? JSON.parse(p.templateData) : {};
   417→        const colorPalette = p.colorPalette ? JSON.parse(p.colorPalette) : null;
   418→
   419→        // Map DB blocks → schema blocks
   420→        const schemaBlocks = (p.blocks || []).map((b) => {
   421→          const content = b.content ? JSON.parse(b.content) : {};
   422→          const layout = b.layout ? JSON.parse(b.layout) : undefined;
   423→          return {
   424→            type: b.blockType,
   425→            ...content,
   426→            ...(layout ? { layout } : {}),
   427→          };
   428→        });
   429→
   430→        const migrated: CanvaPage = {
   431→          id: p.id,
   432→          label: p.label || `Halaman ${p.pageIndex + 1}`,
   433→          bgDataUrl: p.bgImage || null,
   434→          bgColor: p.bgColor || '#ffffff',
   435→          overlay: p.bgOverlay !== null ? Math.round(p.bgOverlay * 100) : 20,
   436→          elements: [], // Always empty — schema is the source of truth
   437→          templateType: (p.templateType as CanvaPage['templateType']) || 'custom',
   438→          colorPalette,
   439→          navConfig,
   440→          templateData,
   441→          templateVariant: (p.variant as 'A' | 'B' | 'C') || undefined,
   442→          contractId: (p.contractId as string) || undefined,
   443→          schema: schema || (schemaBlocks.length > 0 ? { id: p.id, templateType: p.templateType || 'custom', blocks: schemaBlocks } : undefined),
   444→        };
   445→
   446→        return migrated;
   447→      });
   448→
   449→      // Apply schema migration + clear elements[] for schema pages
   450→      const pages = migrateAllPages(rawPages);
   451→
   452→      // D-P0B.3: Apply schema version migrations (v0→v1, etc.) and USE the result.
   453→      const { pages: migratedPages } = migrateAllSchemas(pages);
   454→
   455→      // Strip derived runtime fields from legacy data
   456→      const cleanPages = stripRuntimeFieldsFromPages(migratedPages);
   457→
   458→      // Purity Guard (non-fatal — just logs)
   459→      try {
   460→        for (const page of cleanPages) {
   461→          if (page.schema) {
   462→            assertDocumentPurity(page.schema, `loadFromDB page=${page.id}`);
   463→          }
   464→        }
   465→      } catch (purityErr) {
   466→        logger.warn('CanvaStore', 'Purity check on DB load failed: ' + String(purityErr));
   467→      }
   468→
   469→      // P0-3 Fix: Parse authoring data BEFORE any mutation.
   470→      // If this fails, the entire load is aborted — canva store
   471→      // is NOT mutated, so Project A pages remain intact.
   472→      let parsedAuthoring: Record<string, unknown> | null = null;
   473→      if (data.authoringData) {
   474→        try {
   475→          parsedAuthoring = JSON.parse(data.authoringData) as Record<string, unknown>;
   476→        } catch (err) {
   477→          throw new Error('Failed to parse authoringData: ' + String(err));
   478→        }
   479→      }
   480→
   481→      // ══════════════════════════════════════════════════════════
   482→      // PHASE 3: HYDRATE & COMMIT — Only after ALL parsing succeeds
   483→      // ══════════════════════════════════════════════════════════
   484→      // We've reached this point only if every JSON.parse and
   485→      // migration succeeded. Now it's safe to mutate stores.
   486→
   487→      // Clear runtime caches for the new project data
   488→      clearCompressedHeightCache();
   489→      clearMeasurementCache();
   490→
   491→      // Start hydration — suppresses markDirty() during commit
   492→      useDirtyStore.getState().startHydration();
   493→
   494→      try {
   495→        // Commit canva store
   496→        set({
   497→          pages: cleanPages,
   498→          ratioId: data.ratioId || '16:9',
   499→          currentPageIndex: 0,
   500→          kontenTabRequest: null,
   501→          kontenPanelRequest: false,
   502→          panelRequest: null,
   503→          selectedElId: null,
   504→          selectedElIds: [],
   505→          selectedBlockIds: [],
   506→          leftPanelOpen: true,
   507→          rightPanelOpen: true,
   508→          leftTab: 'pages',
   509→        });
   510→
   511→        // Commit authoring store atomically with canva
   512→        // This prevents the cross-project contamination bug where
   513→        // canva shows Project B but authoring still has Project A.
   514→        //
   515→        // Patch-4 P0-1 Fix: NEVER use current store values as fallback.
   516→        // When loading Project B, every field must come from B's data or
   517→        // be reset to empty defaults. Falling back to store.cp etc. would
   518→        // silently carry Project A's data into Project B — cross-project
   519→        // contamination even when B's authoringData is null or partial.
   520→        if (parsedAuthoring) {
   521→          // eslint-disable-next-line @typescript-eslint/no-explicit-any -- parsedAuthoring is dynamic JSON from DB
   522→          const auth = parsedAuthoring as any;
   523→          useAuthoringStore.setState({
   524→            // Non-schema fields — use B's data or empty default, NEVER store.cp (Project A)
   525→            cp: auth.cp && typeof auth.cp === 'object' && !Array.isArray(auth.cp) ? auth.cp : { ...DEFAULT_CP },
   526→            atp: auth.atp && typeof auth.atp === 'object' && !Array.isArray(auth.atp) ? auth.atp : { ...DEFAULT_ATP },
   527→            petunjuk: auth.petunjuk && typeof auth.petunjuk === 'object' && !Array.isArray(auth.petunjuk) ? auth.petunjuk : { ...DEFAULT_PETUNJUK },
   528→            penutup: auth.penutup && typeof auth.penutup === 'object' && !Array.isArray(auth.penutup) ? auth.penutup : { ...DEFAULT_PENUTUP },
   529→            suara: auth.suara && typeof auth.suara === 'object' && !Array.isArray(auth.suara) ? auth.suara : { ...DEFAULT_SUARA },
   530→            dirty: false,
   531→            // Schema-backed fields — already loaded via deriveProjectionFromPages()
   532→          });
   533→        } else {
   534→          // No authoring data at all — reset ALL non-schema fields to defaults.
   535→          // This prevents Project A's authoring data from leaking into Project B.
   536→          useAuthoringStore.setState({
   537→            cp: { ...DEFAULT_CP },
   538→            atp: { ...DEFAULT_ATP },
   539→            petunjuk: { ...DEFAULT_PETUNJUK },
   540→            penutup: { ...DEFAULT_PENUTUP },
   541→            suara: { ...DEFAULT_SUARA },
   542→            dirty: false,
   543→          });
   544→        }
   545→
   546→        return true;
   547→      } finally {
   548→        // ALWAYS end hydration, even if set() throws (unlikely but defensive).
   549→        useDirtyStore.getState().endHydration();
   550→      }
   551→    } catch (err) {
   552→      // ALL parsing errors land here. NO stores have been mutated
   553→      // because we only mutate in Phase 3 which is unreachable on error.
   554→      logger.warn('CanvaStore', 'Failed to load from DB: ' + String(err));
   555→      return false;
   556→      // NOTE: No endHydration() here because startHydration() is only
   557→      // called in Phase 3. If we reach this catch, hydration was never
   558→      // started, so there's nothing to end. The outer hydration (from
   559→      // ProjectManager) is managed by ProjectManager's finally block.
   560→    }
   561→  },
